local function a(c,al) return {c[1],c[2],c[3],(c[4] or 1)*(al or 1)*theme.opacity} end
local function mix(c1,c2,t) return {c1[1]+(c2[1]-c1[1])*t,c1[2]+(c2[2]-c1[2])*t,c1[3]+(c2[3]-c1[3])*t,(c1[4] or 1)+((c2[4] or 1)-(c1[4] or 1))*t} end
local function clamp(v,lo,hi) return v<lo and lo or (v>hi and hi or v) end
local function ease_out(t) t=clamp(t,0,1); local f=1-t; return 1-f*f*f end
local function hsv(h,s,v) h=(h%1)*6; local i=math.floor(h); local f=h-i; local p=v*(1-s); local q=v*(1-s*f); local u=v*(1-s*(1-f))
   if i==0 then return{v,u,p,1} elseif i==1 then return{q,v,p,1} elseif i==2 then return{p,v,u,1}
   elseif i==3 then return{p,q,v,1} elseif i==4 then return{u,p,v,1} else return{v,p,q,1} end end
local function to_hsv(c) local r,g,b=c[1],c[2],c[3]; local mx=math.max(r,g,b); local mn=math.min(r,g,b); local d=mx-mn; local h=0
   if d>0 then if mx==r then h=((g-b)/d)%6 elseif mx==g then h=(b-r)/d+2 else h=(r-g)/d+4 end h=h/6 end
   return h, (mx==0 and 0 or d/mx), mx end

local COL_MEM={}
local COL_SCR={1,1,1,1}
local function hue_or(ch,base,al)
   al=(al or 1)*theme.opacity
   if ch then
     local c=COL_SCR
     c[1]=ch[1]; c[2]=ch[2]; c[3]=ch[3]; c[4]=(base[4] or 1)*al
     return c
   end
   local m=COL_MEM[base]
   if not m then m={}; COL_MEM[base]=m end
   local c=m[al]
   if not c then
     c={base[1],base[2],base[3],(base[4] or 1)*al}
     m[al]=c
   end
   return c
end

local function approach(k,target,spd) local v=anim[k]; if v==nil then v=target end

local SCR={w=1920,h=1080,f=-1}
local function scr()
   if SCR.f~=FRAME then
     local w,h=draw.GetScreenSize()
     SCR.w,SCR.h,SCR.f=w or 1920,h or 1080,FRAME
   end
   return SCR.w,SCR.h
end
local function mouse_pos() local x,y=utility.GetMousePos(); return x or 0, y or 0 end

local NOW={v=0,f=-1}
local function now()
   if NOW.f~=FRAME then
     NOW.v=utility.GetTime and utility.GetTime() or os.clock()
     NOW.f=FRAME
   end
   return NOW.v
end

local function ekey(vk)
   local ok,r=pcall(input.IsKeyDown,vk)
   if not ok or r==nil or r==false or r==0 then return false end
   return true
end

local TS={w={},h={},n=0}
local function ts(s,sz)
   sz=sz or 12
   local e=TS.w[sz]
   if not e then e={}; TS.w[sz]=e end
   local w=e[s]
   if w==nil then
     local h
     w,h=draw.GetTextSize(s,sz)
     if w==nil then w=0 end
     e[s]=w
     if h then TS.h[sz]=h end
     TS.n=TS.n+1
     if TS.n>4000 then TS.w={}; TS.n=0 end
   end
   return w, TS.h[sz] or 12
end
local function tmid(x,y,h,s,c,sz) sz=sz or 12; local _,th=ts(s,sz); draw.Text(x,y+(h-th)*0.5,s,c,sz) end
local function trt(xr,y,h,s,c,sz) sz=sz or 12; local tw,th2=ts(s,sz); draw.Text(xr-tw,y+(h-th2)*0.5,s,c,sz) return tw end
local function inside(x,y,w,h) local m=ui.m return m.x>=x and m.x<=x+w and m.y>=y and m.y<=y+h end

CFG.last=0
CFG.save=function(slot,quiet)
   slot=tonumber(slot) or tonumber(cfg.cfg_slot) or 1
   local data=nil
   local hs=CFG.http()
   if hs then
     local t={v={},b={},c={}}
     for k,v in pairs(cfg) do if k~="cfg_slot" and k:sub(1,7)~="gun_sgc" then t.v[k]=v end end
     for k,v in pairs(bind) do t.b[k]=v end
     for k,cv in pairs(col) do t.c[k]={cv[1] or 0,cv[2] or 0,cv[3] or 0,cv[4] or 1} end
     local ok,s=pcall(function() return hs:JSONEncode(t) end)
     if ok and type(s)=="string" and #s>0 then data=s end
   end
   if not data then
     local o={"#egofust-config-v1"}
     for k,v in pairs(cfg) do
       if k~="cfg_slot" and k:sub(1,7)~="gun_sgc" then o[#o+1]="v "..k.."="..(type(v)=="boolean" and (v and "1" or "0") or tostring(v)) end
     end
     for k,v in pairs(bind) do o[#o+1]="b "..k.."="..tostring(v) end
     for k,cv in pairs(col) do o[#o+1]="c "..k.."="..(cv[1] or 0)..","..(cv[2] or 0)..","..(cv[3] or 0)..","..(cv[4] or 1) end
     data=table.concat(o,"\n")
   end
   local g=CFG.env()
   if g then g.EGOFUST_CFGS[slot]=data; g.EGOFUST_SLOT=slot end
   local disk=false
   local ok,f=pcall(function() return io.open("egofust_config_"..slot..".cfg","w") end)
   if ok and f then
     local wok=pcall(function() f:write(data) end)
     pcall(function() f:close() end)
     disk=wok
   end
   cfg_dirty=false
   CFG.last=now()
   if not quiet then
     if disk then pcall(function() notify.Success("Config saved","slot "..slot.." - egofust_config_"..slot..".cfg",3) end)
     else pcall(function() notify.Warning("Config saved to session","io.open is blocked, no file written",4) end) end
   end
   return disk
end
CFG.load=function(slot,quiet)
   slot=tonumber(slot) or tonumber(cfg.cfg_slot) or 1
   local g=CFG.env()
   local data=nil
   local ok,f=pcall(function() return io.open("egofust_config_"..slot..".cfg","r") end)
   if ok and f then
     local rok,s=pcall(function() return f:read("*a") end)
     pcall(function() f:close() end)
     if rok and type(s)=="string" and #s>0 then data=s end
   end
   if not data and g and type(g.EGOFUST_CFGS)=="table" then data=g.EGOFUST_CFGS[slot] end
   if not data then
     if not quiet then pcall(function() notify.Warning("Nothing to load","slot "..slot.." is empty",3) end) end
     return false
   end
   local applied=false
   if data:sub(1,1)=="{" then
     local hs=CFG.http()
     if hs then
       local jok,t=pcall(function() return hs:JSONDecode(data) end)
       if jok and type(t)=="table" then
         for k,v in pairs(t.v or {}) do
           if cfg[k]~=nil and k:sub(1,7)~="gun_sgc" then
             if type(v)=="boolean" then cfg[k]=v
             else local x=tonumber(v); if x then cfg[k]=x end end
           end
         end
         for k,v in pairs(t.b or {}) do
           if bind[k]~=nil then local x=tonumber(v); if x then bind[k]=x end end
         end
         for k,v in pairs(t.c or {}) do
           if col[k]~=nil and type(v)=="table" then
             local r1,g1,b1=tonumber(v[1]),tonumber(v[2]),tonumber(v[3])
             if r1 and g1 and b1 then col[k]={r1,g1,b1,tonumber(v[4]) or 1} end
           end
         end
         applied=true
       end
     end
   else
     for line in data:gmatch("[^\n]+") do
       local kind,k,val=line:match("^(%a) ([^=]+)=(.*)$")
       if kind then
         if kind=="v" and cfg[k]~=nil and k:sub(1,7)~="gun_sgc" then
           if type(cfg[k])=="boolean" then cfg[k]=(val=="1" or val=="true")
           else local x=tonumber(val); if x then cfg[k]=x end end
         elseif kind=="b" and bind[k]~=nil then
           local x=tonumber(val); if x then bind[k]=x end
         elseif kind=="c" and col[k]~=nil then
           local r,g1,b1,a1=val:match("([%d%.%-]+),([%d%.%-]+),([%d%.%-]+),([%d%.%-]+)")
           if r then col[k]={tonumber(r),tonumber(g1),tonumber(b1),tonumber(a1) or 1} end
         end
       end
     end
     applied=true
   end
   if cfg.aim_range and cfg.aim_range>1500*M_PER_STUD then cfg.aim_range=1500*M_PER_STUD end
   if cfg.esp_maxdist and cfg.esp_maxdist>1500*M_PER_STUD then cfg.esp_maxdist=1500*M_PER_STUD end
   if cfg.npcs_max_dist and cfg.npcs_max_dist>300*M_PER_STUD then cfg.npcs_max_dist=300*M_PER_STUD end
   if cfg.rad_range and cfg.rad_range>500*M_PER_STUD then cfg.rad_range=500*M_PER_STUD end
   if cfg.plants_range and cfg.plants_range>125*M_PER_STUD then cfg.plants_range=125*M_PER_STUD end
   if cfg.pick_range and cfg.pick_range>75*M_PER_STUD then cfg.pick_range=75*M_PER_STUD end
   if cfg.veh_range and cfg.veh_range>1000*M_PER_STUD then cfg.veh_range=1000*M_PER_STUD end
   if cfg.nodes_range and cfg.nodes_range>225*M_PER_STUD then cfg.nodes_range=225*M_PER_STUD end
   if cfg.bd_range and cfg.bd_range>300*M_PER_STUD then cfg.bd_range=300*M_PER_STUD end
   if cfg.loot_range and cfg.loot_range>225*M_PER_STUD then cfg.loot_range=225*M_PER_STUD end
   if cfg.base_range and cfg.base_range>175*M_PER_STUD then cfg.base_range=175*M_PER_STUD end
   if g then g.EGOFUST_SLOT=slot end
   cfg_dirty=false
   if not quiet then
     if applied then pcall(function() notify.Success("Config loaded","slot "..slot,3) end)
     else pcall(function() notify.Error("Config is corrupted","slot "..slot.." could not be parsed",3) end) end
   end
   return applied
end

local ui={ open=true, menu_edge=false, esp_edge=false, aim_edge=false, appear=1 }

local WIN=nil

local function row_visible(it) return it.show==nil or it.show() end

-- Try common method names so this works across Vector builds.
local function call(obj, names, ...)
   for i=1,#names do
     local fn=obj[names[i]]
     if type(fn)=="function" then
       local ok,r=pcall(fn,obj,...)
       if ok then return r end
     end
   end
   return nil
end

local function set_visible(widget,on)
   call(widget,{"SetVisible","setVisible","SetHidden","setHidden"},on)
   -- some builds invert SetHidden; if both exist prefer SetVisible (tried first)
end

local function refresh_section(sec_list)
   for _,entry in ipairs(sec_list) do
     set_visible(entry.widget, row_visible(entry.it))
   end
end

local function refresh_all()
   for _,lst in pairs(SEC_WIDGETS) do refresh_section(lst) end
end

local function make_color_picker(parent,label,c,on_change)
   local cp=call(parent,{"AddColorPicker","addColorPicker","CreateColorPicker"},label,c[1] or 1,c[2] or 1,c[3] or 1,c[4] or 1)
   if cp then
     local ok=pcall(function()
       cp.SetCallback and cp:SetCallback(on_change)
       or cp.set_callback and cp:set_callback(on_change)
       or cp.OnChange and cp:OnChange(on_change)
     end)
   end
   return cp
end

local function build_group(sec,group)
   local lst={}
   for _,it in ipairs(group.items) do
     local w
     if it.t=="toggle" then
       w=call(sec,{"AddToggle","addToggle","CreateToggle"},it.label,cfg[it.id])
       if w then pcall(function()
         w.SetCallback and w:SetCallback(function(v)
            cfg[it.id]=v; cfg_dirty=true; refresh_all()
         end)
       end) end
       if it.color then
         local cid=it.color
         local cc=col[cid] or {1,1,1,1}
         local cp=make_color_picker(sec,it.label.." Color",cc,function(r,g,b,al)
            col[cid]={r or cc[1],g or cc[2],b or cc[3],al or cc[4]}
            cfg_dirty=true
         end)
         if cp then lst[#lst+1]={it=it,widget=cp} end
       end
     elseif it.t=="slider" then
       w=call(sec,{"AddSlider","addSlider","CreateSlider"},it.label,it.min,it.max,math.floor(cfg[it.id] or it.min),it.suf or "")
       if w then pcall(function()
         w.SetCallback and w:SetCallback(function(v)
            cfg[it.id]=v; cfg_dirty=true
         end)
       end) end
     elseif it.t=="combo" then
       w=call(sec,{"AddComboBox","addComboBox","CreateComboBox","AddDropdown","addDropdown"},it.label,it.items,cfg[it.id] or 1)
       if w then pcall(function()
         w.SetCallback and w:SetCallback(function(idx)
            cfg[it.id]=idx; cfg_dirty=true
            if it.id=="cfg_slot" then
              if CFG.load(idx,true) then pcall(function() notify.Success("Slot "..idx.." loaded","",2) end)
              else pcall(function() notify.Warning("Slot "..idx.." is empty","settings are unchanged",2) end) end
            end
            refresh_all()
         end)
       end) end
     elseif it.t=="bind" then
       w=call(sec,{"AddBind","addBind","CreateBind","AddKeybind","addKeybind"},it.label,vkname(bind[it.id] or 0))
       if w then pcall(function()
         w.SetCallback and w:SetCallback(function(vk)
            bind[it.id]=vk or 0; cfg_dirty=true
         end)
       end) end
     elseif it.t=="color" then
       local cc=col[it.id] or {1,1,1,1}
       w=make_color_picker(sec,it.label,cc,function(r,g,b,al)
         col[it.id]={r or 1,g or 1,b or 1,al or 1}
         cfg_dirty=true
         if it.id=="c_accent" then theme.accent={r or 1,g or 1,b or 1,1} end
       end)
     elseif it.t=="button" then
       w=call(sec,{"AddButton","addButton","CreateButton"},it.label)
       if w then pcall(function()
         w.SetCallback and w:SetCallback(function()
            if it._cb then it._cb() end
            if it.id=="cfg_reset" then refresh_all() end
         end)
       end) end
     elseif it.t=="note" then
       w=call(sec,{"AddLabel","addLabel","CreateLabel","AddText","addText"},it.label)
     end
     if w then
       lst[#lst+1]={it=it,widget=w}
       -- initial visibility
       set_visible(w,row_visible(it))
     end
   end
   SEC_WIDGETS[group]=lst
end

local function build_gui()
   if not gui then return false end
   local win_opts={ name="Ego - Fust Survival" }
   WIN=call(gui,{"CreateWindow","createWindow","NewWindow"},win_opts)
        or call(gui,{"AddWindow","addWindow"})
   if not WIN then return false end
   for i,tab in ipairs(SCHEMA) do
     local t=call(WIN,{"AddTab","addTab","CreateTab","createTab"},tab.name)
     if t then
       TABS[i]=t
       for _,g in ipairs(tab.groups) do
         local sec=call(t,{"AddSection","addSection","CreateSection","createSection","AddGroup","addGroup"},g.name)
         if sec then build_group(sec,g) end
       end
     end
   end
   return true
end

build_gui()

local function toggle_window()
   if not WIN then return end
   local vis=call(WIN,{"IsVisible","isVisible","GetVisible","getVisible"})
   if type(vis)~="boolean" then vis=ui.open end
   local nv=not vis
   local ok=pcall(function()
     WIN.SetVisible and WIN:SetVisible(nv)
     or WIN.setVisible and WIN:setVisible(nv)
     or WIN.Toggle and WIN:Toggle()
   end)
   if not ok then ui.open=nv end
   ui.open=nv
end

local function update_input()
   ui.m.x,ui.m.y=mouse_pos()
   local d=ekey(0x01); ui.m.click=d and not ui.m.was; ui.m.down=d; ui.m.was=d

   local mk=bind.menu_key or 0xA1
   local held=ekey(mk)
   if held and not ui.menu_edge then toggle_window() end
   ui.menu_edge=held

PF.team.sync=function()
   local t=PF.team
   t.setup=true
   pcall(function()
     local rs=nil
     if game then
       rs=game.ReplicatedStorage
       if not rs and game.GetService then rs=game:GetService("ReplicatedStorage") end
     end
     if not rs then return end
     local child=PF.team.child
     local fust=child(rs,"Fust")
     if not fust then return end
     local rem=child(fust,"Remotes")
     if not rem then return end
     local function take(msg,clear)
       if type(msg)~="table" then return end
       local team=msg.team
       if type(team)~="table" then team=msg end
       if type(team.members)~="table" then
         if clear then t.roster={} end
         return
       end
       local ids={}
       for _,m in ipairs(team.members) do
         if type(m)=="table" and type(m.userId)=="number" then ids[m.userId]=true end
       end
       t.roster=ids
     end
     local sync=child(rem,"TeamSync")
     if sync then
       local sig=sync.OnClientEvent
       if not sig then sig=sync.on_client_event end
       local fn=sig and sig.Connect
       if type(fn)~="function" then fn=sig and sig.connect end
       if type(fn)=="function" then
         pcall(fn,sig,function(msg)
           if type(msg)=="table" and msg.kind=="roster" then take(msg,true) end
         end)
       end
     end
     local act=child(rem,"TeamAction")
     if act then
       local inv=act.InvokeServer
       if type(inv)~="function" then inv=act.invoke_server end
       if type(inv)=="function" then
         local function seed()
           local okr,res=pcall(inv,act,{op="sync"})
           if okr and type(res)=="table" then
             take(type(res.roster)=="table" and res.roster or res,false)
           end
         end
         local ok=pcall(function() task.spawn(seed) end)
         if not ok then
           pcall(function() coroutine.wrap(seed)() end)
         end
       end
     end
   end)
end

PF.chains=function(links)
   local adj={}
   for i=1,#links do
     local a,b=links[i][1],links[i][2]
     local t=adj[a]; if not t then t={}; adj[a]=t end
     t[#t+1]=b
     t=adj[b]; if not t then t={}; adj[b]=t end
     t[#t+1]=a
   end
   local deg={}
   for k,t in pairs(adj) do deg[k]=#t end
   local used={}
   local out={}
   local function ekey(a,b) if a>b then a,b=b,a end return a*100+b end
   for i=1,#links do
     local a,b=links[i][1],links[i][2]
     local ek=ekey(a,b)
     if not used[ek] then
       used[ek]=true
       local path={a,b}
       local prev,cur=a,b
       while deg[cur]==2 do
         local nb=adj[cur]
         local nxt=nb[1]==prev and nb[2] or nb[1]
         local k2=ekey(cur,nxt)
         if used[k2] then break end
         used[k2]=true
         path[#path+1]=nxt
         prev,cur=cur,nxt
       end
       prev,cur=b,a
       while deg[cur]==2 do
         local nb=adj[cur]
         local nxt=nb[1]==prev and nb[2] or nb[1]
         local k2=ekey(cur,nxt)
         if used[k2] then break end
         used[k2]=true
         table.insert(path,1,nxt)
         prev,cur=cur,nxt
       end
       local pts={}
       for k=1,#path do pts[k]={0,0} end
       out[#out+1]={nodes=path,pts=pts}
     end
   end
   return out
end

local function esp_mate(p,lp)
   if not p then return false end
   local id=PF.pflag(p).uid
   if not id then return false end
   if lp and lp.UserId==id then return true end
   local t=PF.team
   if not t.setup then pcall(t.sync) end
   if t.roster[id] then return true end
   if FRAME>=t.next then
     t.next=FRAME+60
     local ids=t.ids
     for k in pairs(ids) do ids[k]=nil end
     local tf
     pcall(function()
       local ws
       if game then
         ws=game.Workspace or game.workspace
         if not ws and game.GetService then ws=game:GetService("Workspace") end
       end
       if not ws then ws=workspace end
       if not ws then return end
       local child=PF.team.child
       local cw=nil
       local okc,cv=pcall(PF.get,ws,"CurrentCamera")
       if okc and cv then cw=cv end
       if not cw then cw=child(ws,"Camera") end
       tf=child(cw,"FustTeamTags")
       if not tf then tf=child(ws,"FustTeamTags") end
       if tf then
         local list
         local fn=tf.GetChildren
         if type(fn)=="function" then
           local ok,v=pcall(fn,tf)
           if ok and type(v)=="table" then list=v end
         end
         if not list then
           fn=tf.get_children
           if type(fn)=="function" then
             local ok,v=pcall(fn,tf)
             if ok and type(v)=="table" then list=v end
           end
         end
         if list then
           for _,ch in ipairs(list) do
             local nm=ch and ch.Name
             if nm and nm:sub(1,3)=="Tag" then
               local uid=tonumber(nm:sub(4))
               if uid then ids[uid]=true end
             end
           end
         else
           if child(tf,"Tag"..id) then ids[id]=true end
         end
       end
     end)
     t.tf=tf
   end
   if t.ids[id] then return true end
   if t.tf then return false end
   if not (lp and lp.HasTeam and p.HasTeam) then return false end
   return p.Team==lp.Team
end

local function esp_player(p)
   local F=ESP_S.f
   local fl=PF.pflag(p)
   if p==F.lp or fl.il then return end
   if fl.al==false then return end
   local pos=PF.pmemo(p)
   if not pos then return end
   local dx,dy,dz=pos.X-F.lx,pos.Y-F.ly,pos.Z-F.lz
   local d2=dx*dx+dy*dy+dz*dz
   if d2>F.md2 then return end
   if F.dot then
     local ax,ay,az
     if F.cx then ax,ay,az=pos.X-F.cx,pos.Y-F.cy,pos.Z-F.cz
     else ax,ay,az=dx,dy,dz end
     if ax*F.fx+ay*F.fy+az*F.fz<=0 and (F.cx or d2>2500) then return end
   end
   if F.team and esp_mate(p,F.lp) then return end
   local dist=math.sqrt(d2)
   local ch=nil
   if F.chroma then ch=PF.hue_fill(ESP_S.ch,(F.chue+F.idx*0.07)%1,0.75,1) end
   local c_box=col.c_box or {0.45,0.78,1,1}
   local c_bone=col.c_enemy or {0.95,0.35,0.4,1}
   local c_name=col.c_name or {1,1,1,1}
   local c_dist=col.c_dist or {0.87,0.93,1,1}
   local c_weapon=col.c_weapon or c_name
   local c_tr=col.c_tracer or c_bone
   local c_hat=col.c_hat or c_name
   local tsz=F.tsz
   local b
   if cfg.esp_box or cfg.esp_hp or cfg.esp_dist or cfg.esp_name
     or cfg.esp_weapon or cfg.esp_tracer then
     local okb,bv=pcall(PF.bounds,p); if okb then b=bv end
   end
   local bx,by,bw,bh
   if b and b.valid and (b.w or 0)>1 and (b.h or 0)>1 then bx,by,bw,bh=b.x,b.y,b.w,b.h end
   if b and b.valid==false then return end
   local big = (bh==nil) or bh>=20
   local pname,ptool
   if cfg.esp_name or cfg.esp_hp then
     local okn,nv=pcall(PF.name,p); if okn then pname=nv end
   end
   if cfg.esp_weapon then
     local okt,tv=pcall(PF.tool,p); if okt then ptool=tv end
   end
   local head_x,head_y
   if cfg.esp_hat or cfg.esp_headdot or not bx then
     local okhb,hx,hy,hvis=pcall(PF.bonescreen,p,"Head")
     if okhb and hx and hy and hvis~=false and hx==hx and hy==hy then head_x,head_y=hx,hy end
   end
   local pps=(bh and bh>0) and bh/5.6 or 0
   if cfg.esp_hat or cfg.esp_headdot then
     local okh,hv=pcall(PF.head,p)
     if okh and hv then
       local x0,y0,v0=draw.WorldToScreen(hv.X,hv.Y,hv.Z)
       local y2,v2=select(2,draw.WorldToScreen(hv.X,hv.Y+1,hv.Z))
       if v0 and v2 and y0==y0 and y2==y2 then
         local sc=math.abs(y2-y0)
         if sc>0 then pps=sc end
       end
       if not head_x and v0 and x0==x0 and y0==y0 then head_x,head_y=x0,y0 end
     end
   end
   local hat_apex
   local hat_hs=0
   local hat_sq=1
   if cfg.esp_hat and head_x and pps>0 then
     hat_hs=(cfg.esp_hat_size or 100)/100
     hat_apex=head_y-1.16*hat_hs*pps
     if F.cy then
       local ex,ey,ez=pos.X-F.cx,(pos.Y+1.5)-F.cy,pos.Z-F.cz
       local el=math.sqrt(ex*ex+ey*ey+ez*ez)
       if el>0.001 then hat_sq=math.abs(ey)/el end
     elseif F.dot then
       hat_sq=math.abs(F.fy)
     end
     if hat_sq<0.25 then hat_sq=0.25 end
     if hat_sq>1 then hat_sq=1 end
   end
   if cfg.esp_box and bx then
     local c=hue_or(ch,c_box,1)
     local x0,y0,w0,h0=bx-2,by-2,bw+4,bh+4
     if cfg.esp_boxstyle==1 and big then
       if draw.CornerBox then draw.CornerBox(x0,y0,w0,h0,c) else draw.Rect(x0,y0,w0,h0,c,0,1) end
     elseif cfg.esp_boxstyle==3 then
       local tex=pic_tex(cfg.esp_picstyle or 1)
       if tex then
         local pop=clamp(cfg.esp_pic_opacity or 100,10,100)/100
         local al=math.floor(clamp(c[4],0,1)*pop*255+0.5)
         local tr,tg,tb=255,255,255
         if ch then
           tr=math.floor(clamp(c[1],0,1)*255+0.5)
           tg=math.floor(clamp(c[2],0,1)*255+0.5)
           tb=math.floor(clamp(c[3],0,1)*255+0.5)
         end
         if al>0 then draw.Image(tex,x0,y0,w0,h0,tr,tg,tb,al) end
       else
         draw.Rect(x0,y0,w0,h0,c,0,1)
       end
     else draw.Rect(x0,y0,w0,h0,c,0,1) end
   end
   if cfg.esp_hp and bx then
     local hh,mh=0,100
     local he
     if pname then
       he=ESP_S.hp[pname]
       if not he then he={hp=100,mhp=100,next=0}; ESP_S.hp[pname]=he end
       if FRAME>=he.next then
         local okh2,hv2=pcall(PF.hp,p); if okh2 and hv2 then he.hp=hv2 end
         local okm2,mv2=pcall(PF.mhp,p); if okm2 and mv2 then he.mhp=mv2 end
         he.next=FRAME+3
       end
       hh,mh=he.hp or 0,he.mhp or 100
     else
       local okh2,hv2=pcall(PF.hp,p); if okh2 and hv2 then hh=hv2 end
       local okm2,mv2=pcall(PF.mhp,p); if okm2 and mv2 then mh=mv2 end
     end
     local gap=clamp(bw*0.12,1.5,5)
     pcall(draw.HealthBar,bx-gap,by,bh,hh,mh)
     if cfg.esp_hptext then
       local fsz=math.max(8,tsz-3)
       local frac=clamp(mh>0 and hh/mh or 0,0,1)
       local ht=string.format("%d",math.floor(hh+0.5))
       local tw2,th2=ts(ht,fsz)
       local hc=HP_RAMP[math.floor(frac*20+0.5)]
       draw.Text(math.max(1,bx-gap-10-tw2),by+(bh-th2)*0.5,ht,hue_or(ch,hc,1),fsz)
     end
   end
   local anchor_x,anchor_y=head_x,head_y
   local name_y=by and (by-tsz-4) or (anchor_y and anchor_y-tsz-6)
   if hat_apex and (not name_y or hat_apex<name_y) then name_y=hat_apex-tsz-4 end
   local cx = bx and (bx+bw*0.5) or anchor_x
   if cfg.esp_name and pname and cx and name_y then
     local tw=esp_tw(pname,tsz)
     draw.Text(cx-tw*0.5,name_y,pname,hue_or(ch,c_name,1),tsz)
   end
   local stack_y=by and (by+bh+3) or (anchor_y and anchor_y+4)
   if cfg.esp_dist and cx and stack_y then
     local dm=math.floor(dist)
     local de=ESP_S.dt[dm]
     if not de then de={}; ESP_S.dt[dm]=de end
     local dl=de.txt
     if not dl then dl=dm.."m"; de.txt=dl end
     local dsz=tsz-2
     local dw=de[dsz]
     if not dw then dw=ts(dl,dsz); de[dsz]=dw end
     draw.Text(cx-dw*0.5,stack_y,dl,hue_or(ch,c_dist,0.9),dsz)
     stack_y=stack_y+dsz+1
   end
   if cfg.esp_weapon and cx and stack_y and ptool and ptool~="" then
     local wsz=tsz-2
     local tw=esp_tw(ptool,wsz)
     draw.Text(cx-tw*0.5,stack_y,ptool,hue_or(ch,c_weapon,0.9),wsz)
   end
   if cfg.esp_tracer then
     local oy=(cfg.esp_tracer_from==1) and F.sh or ((cfg.esp_tracer_from==3) and 0 or F.sh*0.5)
     local tx,ty
     if bx then tx,ty=bx+bw*0.5,by+bh elseif anchor_x then tx,ty=anchor_x,anchor_y end
     if tx then draw.Line(F.hw,oy,tx,ty,hue_or(ch,c_tr,1),1) end
   end
   if cfg.esp_skel and (bx or head_x) and (pps<=0 or pps>=1.0) then
     local px,py,pok=PF.px,PF.py,PF.pok
     local names=PF.bone_names
     local d=PF.bonesmemo(p)
     for i=1,#names do
       local e=d and d[names[i]]
       if e and e[1] and e[2] then
         local vx,vy=e[1],e[2]
         if vx>-8 and vy>-8 and vx<F.sw+8 and vy<F.sh+8 then
           px[i],py[i],pok[i]=vx,vy,true
         else
           pok[i]=nil
         end
       else
         pok[i]=nil
       end
     end
     local c=hue_or(ch,c_bone,0.9)
     local links=PF.bone_links
     for li=1,#links do
       local l=links[li]
       local a,b=l[1],l[2]
       if pok[a] and pok[b] then
         draw.Line(px[a],py[a],px[b],py[b],c,1)
       end
     end
   end
   if cfg.esp_headdot and head_x then
     local hs=(cfg.esp_headdot_size or 100)/100
     local hr=pps>0 and (0.62*hs*pps) or (bx and bh*0.11*hs or nil)
     if hr and hr>0.4 then
       local c=hue_or(ch,c_bone,1)
       if hr<2.4 then
         draw.CircleFilled(head_x,head_y,math.max(hr,1.6),c,10)
       else
         draw.Circle(head_x,head_y,hr,c,math.max(10,math.floor(hr*3)),1)
       end
     end
   end
   if hat_apex and pps>0 then
     local rad=0.62*hat_hs*pps
     local brimy=head_y-0.44*hat_hs*pps
     local c=hue_or(ch,c_hat,1)
     local sq=hat_sq
     if rad<3 then
       if draw.PolyFilled then
         draw.PolyFilled({{head_x,hat_apex},{head_x+rad,brimy},{head_x-rad,brimy}},c)
       else
         draw.Line(head_x,hat_apex,head_x+rad,brimy,c,1)
         draw.Line(head_x+rad,brimy,head_x-rad,brimy,c,1)
         draw.Line(head_x-rad,brimy,head_x,hat_apex,c,1)
       end
     else
       local hc,hsn=PF.hc,PF.hs
       local xs,ys=PF.xs,PF.ys
       local seg=16
       for i=1,seg do
         xs[i]=head_x+hc[i]*rad
         ys[i]=brimy+hsn[i]*rad*sq
       end
       local pc=draw.PolyClosed
       local ring=PF.ring
       if type(pc)=="function" then
         for i=1,seg do
           local p=ring[i]
           if not p then p={0,0}; ring[i]=p end
           p[1]=xs[i]; p[2]=ys[i]
         end
         for i=seg+1,#ring do ring[i]=nil end
         pc(ring,c,1)
       else
         for i=1,seg do local j=i%seg+1; draw.Line(xs[i],ys[i],xs[j],ys[j],c,1) end
       end
       for i=1,seg,2 do draw.Line(head_x,hat_apex,xs[i],ys[i],c,1) end
     end
   end
end

local function chams_one(p)
   local F=ESP_S.c
   local fl=PF.pflag(p)
   if p==F.lp or fl.il or fl.al==false then return end
   local pos=PF.pmemo(p)
   if not pos then return end
   local dx,dy,dz=pos.X-F.lx,pos.Y-F.ly,pos.Z-F.lz
   local d2=dx*dx+dy*dy+dz*dz
   if d2>F.md2 then return end
   if F.dot then
     local ax,ay,az
     if F.cx then ax,ay,az=pos.X-F.cx,pos.Y-F.cy,pos.Z-F.cz
     else ax,ay,az=dx,dy,dz end
     if ax*F.fx+ay*F.fy+az*F.fz<=0 and (F.cx or d2>2500) then return end
   end
   if F.team and esp_mate(p,F.lp) then return end
   local ch=nil
   if F.chroma then ch=PF.hue_fill(ESP_S.ch,(F.chue+F.idx*0.07)%1,0.8,1) end
   draw.ChamsPlayer(p,hue_or(ch,F.c1,1),F.style)
end

   end
end

PF.radar={rc={1,1,1,1},ch={1,1,1,1}}
PF.lv={}
PF.lvec=function(p)
   local e=PF.lv[p]
   if not e then e={x=0,z=0,next=0}; PF.lv[p]=e end
   if FRAME>=e.next then
     e.next=FRAME+6
     local ok,lv=pcall(PF.get,p,"LookVector")
     if ok and lv then e.x,e.z=lv.X or 0,lv.Z or 0 end
   end
   return e.x,e.z
end

PF.radar_one=function(p,idx)
   local R=PF.radar
   local lp=R.lp
   local fl=PF.pflag(p)
   if p==lp or fl.il then return end
   if fl.al==false then return end
   local mate=esp_mate(p,lp)
   if cfg.esp_team and mate then return end
   local pos=PF.pmemo(p)
   if not pos then return end
   local ox,oy,oz=R.ox,R.oy,R.oz
   local dx,dz=pos.X-ox,pos.Z-oz
   local dd=math.sqrt(dx*dx+dz*dz)
   if dd>R.rng then return end
   local px=R.cx+(dx*R.rtx+dz*R.rtz)*(R.rsz*0.5-8)/R.rng
   local py=R.cy-(dx*R.fx+dz*R.fz)*(R.rsz*0.5-8)/R.rng
   local cb=col.c_enemy or {0.95,0.35,0.4,1}
   local ch=nil
   if R.chroma then ch=PF.hue_fill(R.ch,(R.chue+idx*0.07)%1,0.75,1) end
   local c=hue_or(ch,cb,R.al)
   local ca=mate and c[4]*0.5 or c[4]
   local lx,lz=PF.lvec(p)
   if lx~=0 or lz~=0 then
     local ll=math.sqrt(lx*lx+lz*lz)
     if ll>1e-6 then
       local ex=(lx*R.rtx+lz*R.rtz)/ll
       local ey=-(lx*R.fx+lz*R.fz)/ll
       local r0=mate and 2.2 or 3
       local tc=PF.rct
       tc[1]=c[1]; tc[2]=c[2]; tc[3]=c[3]; tc[4]=ca*0.85
       draw.Line(px+ex*r0,py+ey*r0,px+ex*(r0+7),py+ey*(r0+7),tc,1)
     end
   end
   local rc=R.rc
   rc[1]=c[1]; rc[2]=c[2]; rc[3]=c[3]; rc[4]=ca
   draw.CircleFilled(px,py,mate and 2.2 or 3,rc,10)
   local ddx,ddz=pos.X-ox,pos.Z-oz
   local ddy=pos.Y-oy
   local dm=math.floor(math.sqrt(ddx*ddx+ddy*ddy+ddz*ddz))
   local txt=PF.rdtxt[dm]
   if not txt then txt=dm.."m"; PF.rdtxt[dm]=txt end
   local tw=PF.rdw[dm]
   if not tw then tw=ts(txt,9); PF.rdw[dm]=tw end
   local tx2=px+11
   if tx2+tw>R.rx+R.rsz-3 then tx2=px-11-tw end
   draw.Text(tx2,py-5,txt,rc,9)
end

local function run_radar()
   if not cfg.rad_on then return end
   local sw,sh=scr()
   local rsz=cfg.rad_size or 160
   local rng=math.max(0.5,cfg.rad_range or 50)/PF.m
   local al=clamp((cfg.rad_alpha or 80)/100,0,1)
   local rx=clamp(sw*(cfg.rad_x or 88)/100-rsz*0.5,0,math.max(0,sw-rsz))
   local ry=clamp(sh*(cfg.rad_y or 8)/100,0,math.max(0,sh-rsz))
   local cx,cy=rx+rsz*0.5,ry+rsz*0.5
   draw.RectFilled(rx-2,ry-2,rsz+4,rsz+4,{0,0,0,0.55*al},8)
   draw.Rect(rx-2,ry-2,rsz+4,rsz+4,{1,1,1,0.10*al},8,1)
   for i=1,3 do draw.Circle(cx,cy,(rsz*0.5-3)*i/3,{1,1,1,0.07*al},40,1) end
   draw.Line(cx,ry+4,cx,ry+rsz-4,{1,1,1,0.07*al},1)
   draw.Line(rx+4,cy,rx+rsz-4,cy,{1,1,1,0.07*al},1)
   draw.CircleFilled(cx,cy,2.6,{0.3,1,0.55,al},10)
   local vw=PF.getview()
   if not vw.cx then return end
   local fx,fz=vw.fx or 0,vw.fz or 0
   local fl=math.sqrt(fx*fx+fz*fz); if fl<1e-6 then fl=1 end
   fx,fz=fx/fl,fz/fl
   local R=PF.radar
   R.lp=vw.lp
   R.rx,R.rsz,R.rng,R.al=rx,rsz,rng,al
   R.cx,R.cy=cx,cy
   R.fx,R.fz=fx,fz
   R.rtx,R.rtz=-fz,fx
   R.ox,R.oy,R.oz=vw.cx,vw.cy,vw.cz
   if vw.lx then R.ox,R.oy,R.oz=vw.lx,vw.ly,vw.lz end
   R.chroma=cfg.esp_chroma
   R.chue=now()*0.12
   local players=PF.players()
   if players then
     for idx,p in ipairs(players) do pcall(PF.radar_one,p,idx) end
   end
   if FRAME%600==0 then
     for k,e in pairs(PF.lv) do
       if e.next<FRAME-600 then PF.lv[k]=nil end
     end
   end
   draw.Text(rx+6,ry+rsz-14,"RADAR",{1,1,1,0.35*al},9)
end

   n=n+1

local function hud_key_rows()

   v=bind.esp_toggle or 0

   if v~=0 then n=hud_krow(rows,n,"Menu",v) end
   for i=#rows,n+1,-1 do rows[i]=nil end
   return rows
end

local function run_hud()
   local sw,sh=scr()
   local apm=approach("hud_mark", cfg.hud_mark and 1 or 0, 17)
   local apk=approach("hud_keys", cfg.hud_keys and 1 or 0, 15)
   local apb=approach("hud_bar",  cfg.hud_bar  and 1 or 0, 16)

   if apm>0.01 then
     local fps=math.floor((utility.GetFPS and utility.GetFPS()) or 0)
     if not fps or fps<=0 then fps=math.floor(1/(_dt>0 and _dt or 0.016)) end
     local up=math.floor(now()-hud_start)
     local ut
     if up~=HUD_S.upv then
       HUD_S.upv=up
       ut = up<60 and string.format("%ds",up)
         or (up<3600 and string.format("%dm",math.floor(up/60)))
         or string.format("%dh%02dm",math.floor(up/3600),math.floor((up%3600)/60))
       HUD_S.ups=ut
     else
       ut=HUD_S.ups
     end
     local ft
     if fps~=HUD_S.fpsv then
       HUD_S.fpsv=fps
       ft=string.format("%d FPS",fps)
       HUD_S.fpss=ft
     else
       ft=HUD_S.fpss
     end
     local lw=ts("Ego",14); local pw2=ts("Fust Survival",11)
     local fw=ts(ft,11); local uw=ts(ut,11)
     local ph=30
     local wpx=14+lw+5+8+6+pw2+10+1+10+fw+10+uw+14
     local wx=(cfg.hud_mark_x or -1)>=0 and cfg.hud_mark_x or (sw-wpx-16)
     local wy=(cfg.hud_mark_y or -1)>=0 and cfg.hud_mark_y or 14
     wx=clamp(wx,0,math.max(0,sw-wpx)); wy=clamp(wy,0,math.max(0,sh-ph))
     if inside(wx,wy,wpx,ph) and ui.m.click then
       cfg.hud_mark_x=wx; cfg.hud_mark_y=wy
       ui.drag_mark={ox=ui.m.x-wx,oy=ui.m.y-wy}
       ui.m.click=false
     end
     if ui.drag_mark then
       cfg.hud_mark_x=clamp(ui.m.x-ui.drag_mark.ox,0,math.max(0,sw-wpx))
       cfg.hud_mark_y=clamp(ui.m.y-ui.drag_mark.oy,0,math.max(0,sh-ph))
       cfg_dirty=true
       if not ui.m.down then ui.drag_mark=nil end
     end
     draw.RectFilled(wx,wy,wpx,ph,{theme.base[1],theme.base[2],theme.base[3],0.9*apm},3)
     draw.Rect(wx,wy,wpx,ph,a(theme.line,0.85*apm),3,1)
     tmid(wx+14,wy,ph,"Ego", a(theme.text,1), 14)
     local dcx,dcy=wx+14+lw+5+4, wy+ph*0.5
     if draw.PolyFilled then pcall(draw.PolyFilled, {{dcx,dcy-4},{dcx+4,dcy},{dcx,dcy+4},{dcx-4,dcy}}, a(theme.accent,1))
     else draw.RectFilled(dcx-3,dcy-3,6,6, a(theme.accent,1), 1) end
     local px=dcx+4+6
     tmid(px,wy,ph,"Fust Survival", a(theme.sub,1), 11)
     local gx=px+pw2+10
     draw.Line(gx,wy+8,gx,wy+ph-8, a(theme.line,0.6*apm), 1)
     local tx=gx+11
     local fc = fps>=100 and theme.ok or (fps>=45 and theme.text or theme.bad)
     tmid(tx,wy,ph,ft, a(fc,1), 11)
     tmid(tx+fw+10,wy,ph,ut, a(theme.faint,1), 11)
   end

   if apk>0.01 then
     local rows=hud_key_rows()
     if #rows>0 then
       local rh=21; local pw=178; local ph=40+#rows*rh
       local kx=(cfg.hud_keys_x or -1)>=0 and cfg.hud_keys_x or (sw-pw-16)
       local ky=(cfg.hud_keys_y or -1)>=0 and cfg.hud_keys_y or (sh-ph-18)

       draw.RectFilled(kx,ky,pw,ph,{theme.base[1],theme.base[2],theme.base[3],0.9*apk},4)
       draw.Rect(kx,ky,pw,ph,a(theme.line,0.85*apk),4,1)
       hud_head(kx+13,ky+11,pw-26,"01","Keybinds")

         local ry=ky+32+(i-1)*rh
         if i%2==0 then draw.RectFilled(kx+9,ry,pw-18,rh, {1,1,1,0.022*apk}, 0) end
         tmid(kx+14,ry,rh,r[1], a(theme.sub,1), 11)

         tmid(cx,cy,ch,r[2], a(theme.text,1), 10)
       end
     end
   end

   if apb>0.01 then
     local feats=HUD_S.feats
     for i=#feats,1,-1 do feats[i]=nil end

     if cfg.esp_on then feats[#feats+1]="ESP" end

       feats[#feats+1]="WORLD"
     end
     if #feats>0 then
       local gap=7; local pad=16; local ph=34
       local widths=HUD_S.w; local tot=pad*2+(#feats-1)*gap
       for i,lb in ipairs(feats) do
         local cw=math.max(48, ts(lb,10)+20); widths[i]=cw; tot=tot+cw
       end
       local bx=(cfg.hud_bar_x or -1)>=0 and cfg.hud_bar_x or (sw*0.5-tot*0.5)
       local by=(cfg.hud_bar_y or -1)>=0 and cfg.hud_bar_y or (sh-ph-18)
       bx=clamp(bx,0,math.max(0,sw-tot)); by=clamp(by,0,math.max(0,sh-ph))

         cfg_dirty=true
         if not ui.m.down then ui.drag_bar=nil end
       end
       draw.RectFilled(bx,by,tot,ph,{theme.base[1],theme.base[2],theme.base[3],0.9*apb},4)
       draw.Rect(bx,by,tot,ph,a(theme.line,0.85*apb),4,1)
       local e=mix(theme.accent,theme.accent2,0.5)
       draw.RectFilled(bx+pad-2,by+ph-2,tot-(pad-2)*2,2, {e[1],e[2],e[3],0.9*apb}, 1)

       for i,lb in ipairs(feats) do

         if i<#feats then
           local sx=px+cw+gap*0.5
           draw.Line(sx,by+9,sx,by+ph-10, a(theme.line,0.5*apb), 1)
         end
         px=px+cw+gap
       end
     end
   end
end

   local c=XH_R
   local dx,dy=x-c.cx,y-c.cy
   return c.cx+dx*c.co-dy*c.si, c.cy+dx*c.si+dy*c.co
end
local function xh_poly(pts,c,th)
   if draw.PolyClosed then pcall(draw.PolyClosed, pts, c, th); return end
   for i=1,#pts do
     local p=pts[i]; local q=pts[i%#pts+1]
     draw.Line(p[1],p[2],q[1],q[2],c,th)
   end
end

local function run_crosshair()
   if not cfg.xh_on then return end
   local sw,sh=scr(); local cx,cy=sw*0.5,sh*0.5
   if cfg.xh_follow and cfg.aim_on and XH_R.tv then cx,cy=XH_R.tx,XH_R.ty end
   local pulse=1
   if cfg.xh_pulse then
     local ph=(XH_R.ph or 0)+(cfg.xh_pulse_speed or 100)/60*math.pi*2*_dt
     ph=ph%(math.pi*2)
     XH_R.ph=ph
     pulse=1+0.35*math.sin(ph)
   end
   local s=(cfg.xh_size or 10)*pulse
   local gp=(cfg.xh_gap or 4)*pulse
   local th=cfg.xh_thick
   local c=hue_or(cfg.xh_chroma and hsv((now()*0.12)%1,0.8,1) or nil,col.c_xh,1); local st=cfg.xh_style
   if cfg.xh_spin then
     xh_ang=(xh_ang+(cfg.xh_speed or 0)*math.pi/180*_dt)%(math.pi*2)
   end
   local ang=cfg.xh_spin and xh_ang or 0
   XH_R.cx,XH_R.cy=cx,cy
   XH_R.co,XH_R.si=math.cos(ang),math.sin(ang)
   local R=xh_rot
   if st==1 then
     local x1,y1=R(cx-gp-s,cy); local x2,y2=R(cx-gp,cy)
     local x3,y3=R(cx+gp,cy);   local x4,y4=R(cx+gp+s,cy)
     local x5,y5=R(cx,cy-gp-s); local x6,y6=R(cx,cy-gp)
     local x7,y7=R(cx,cy+gp);   local x8,y8=R(cx,cy+gp+s)
     draw.Line(x1,y1,x2,y2,c,th); draw.Line(x3,y3,x4,y4,c,th)
     draw.Line(x5,y5,x6,y6,c,th); draw.Line(x7,y7,x8,y8,c,th)
   elseif st==2 then
     local r=math.max(0.9, s*0.32+th*0.5)
     draw.CircleFilled(cx,cy,r,c,math.max(8,math.floor(r*3)))
   elseif st==3 then
     draw.Circle(cx,cy,s,c,24,th)
   elseif st==4 then
     local x1,y1=R(cx-gp-s,cy); local x2,y2=R(cx-gp,cy)
     local x3,y3=R(cx+gp,cy);   local x4,y4=R(cx+gp+s,cy)
     local x5,y5=R(cx,cy+gp);   local x6,y6=R(cx,cy+gp+s)
     draw.Line(x1,y1,x2,y2,c,th); draw.Line(x3,y3,x4,y4,c,th)
     draw.Line(x5,y5,x6,y6,c,th)
   elseif st==5 then
     local r=gp+s
     local h=r*math.sqrt(3)*0.5
     local ux,uy=R(cx,cy-r)
     local lx,ly=R(cx-h,cy+r*0.5); local rx2,ry2=R(cx+h,cy+r*0.5)
     local dx,dy=R(cx,cy+r)
     local l2x,l2y=R(cx-h,cy-r*0.5); local r2x,r2y=R(cx+h,cy-r*0.5)
     xh_poly({{ux,uy},{lx,ly},{rx2,ry2}}, c, th)
     xh_poly({{dx,dy},{l2x,l2y},{r2x,r2y}}, c, th)
   elseif st==6 then
     local r=gp+s
     local ax1,ay1=R(cx-r,cy);  local ax2,ay2=R(cx+r,cy)
     local ax3,ay3=R(cx,cy-r);  local ax4,ay4=R(cx,cy+r)
     draw.Line(ax1,ay1,ax2,ay2,c,th); draw.Line(ax3,ay3,ax4,ay4,c,th)
     local h1x,h1y=R(cx-r,cy-r)
     local h2x,h2y=R(cx+r,cy-r)
     local h3x,h3y=R(cx+r,cy+r)
     local h4x,h4y=R(cx-r,cy+r)
     draw.Line(ax1,ay1,h1x,h1y,c,th); draw.Line(ax3,ay3,h2x,h2y,c,th)
     draw.Line(ax2,ay2,h3x,h3y,c,th); draw.Line(ax4,ay4,h4x,h4y,c,th)
   end
   if cfg.xh_dot and st~=2 then draw.CircleFilled(cx,cy,math.max(0.9, th+s*0.07),c,10) end
end

local W_PART_CACHE={}
local function w_part(inst,key,extra)
   if not inst then return nil end
   if not key then key=W_SWEEP.key(inst) end
   local cached=W_PART_CACHE[key]
   if cached then
     if cached.p then
       cached.gen=W_CACHE.gen
       return cached.p
     end
     if cached.gen==W_CACHE.gen then return nil end
   end
   local p
   if extra then
     local fn=inst.FindFirstChild
     if type(fn)=="function" then
       for i=1,#extra do
         local ok,c=pcall(fn,inst,extra[i])
         if ok and c and W_PART_CLASS[c.ClassName] then p=c; break end
       end
     end
   end
   if not p and W_PART_CLASS[inst.ClassName] then p=inst end
   if not p then
     local okp,v=pcall(PF.prim,inst)
     if okp then p=v end
   end
   if not p then
     local fn=inst.FindFirstChild
     if type(fn)=="function" then
       for i=1,#W_PART_NAMES do
         local ok,c=pcall(fn,inst,W_PART_NAMES[i])
         if ok and c and W_PART_CLASS[c.ClassName] then p=c; break end
       end
     end
   end
   if not p then
     local fn=inst.GetChildren
     if type(fn)=="function" then
       local ok,ch=pcall(fn,inst)
       if ok and type(ch)=="table" then
         for i=1,#ch do
           local c=ch[i]
           if W_PART_CLASS[c.ClassName] then p=c; break end
         end
       end
     end
   end
   W_PART_CACHE[key]={p=p or false,gen=W_CACHE.gen}
   return p
end

local W_FIX={}
local function w_pos(inst,part,fixed,key)
   if not key then key=W_SWEEP.key(inst) end
   if fixed then
     local c=W_FIX[key]
     if c then c.gen=W_CACHE.gen; return c.x,c.y,c.z end
   end
   local p=part or w_part(inst,key)
   if not p then return nil end
   local ok,pos=pcall(PF.pos,p)
   if not (ok and pos) then return nil end
   local x=pos.X or pos.x; local y=pos.Y or pos.y; local z=pos.Z or pos.z
   if not x or not y or not z then return nil end
   if fixed then W_FIX[key]={x=x,y=y,z=z,gen=W_CACHE.gen} end
   return x,y,z
end

local function w_item_name(inst)
   local okn,nm=pcall(PF.name,inst)
   if okn and type(nm)=="string" and nm~="" and not nm:lower():match("^item%d*$") then return nm end
   local txt=W_SWEEP.ptxt(inst)
   if txt then return txt end
   local mk,dk,sk,nk=W_SWEEP.sig(inst)
   local tpl=(mk and W_CACHE.itemmesh[mk]) or (dk and W_CACHE.itemdist[dk])
     or (sk and W_CACHE.itemsz[sk]) or (nk and W_CACHE.itemnm[nk])
   if tpl then
     tpl=tpl:gsub("_deathtax$","")

     if alias then return alias end
     local pretty=tpl:gsub("^Icon",""):gsub("(%l)(%u)","%1 %2"):gsub("(%u)(%u%l)","%1 %2")
     if #pretty>0 then return pretty end
   end
   return "Dropped Item"
end

local function w_children(folder)
   if not folder then return nil end

     if ok and type(ch)=="table" then return ch end
   end
   fn=folder.get_children
   if type(fn)=="function" then
     local ok,ch=pcall(fn,folder)
     if ok and type(ch)=="table" then return ch end
   end
   return nil
end

local function w_folder(ws,name)
   if not ws then return nil end
   local fn=ws.FindFirstChild
   if type(fn)=="function" then
     local ok,f=pcall(fn,ws,name)
     if ok and f then return f end
   end
   fn=ws.find_first_child
   if type(fn)=="function" then
     local ok,f=pcall(fn,ws,name)
     if ok and f then return f end
   end
   local okg,f=pcall(PF.get,ws,name)
   if okg and f then return f end
   local ch=W_SWEEP.wsch
   if ch==nil then ch=w_children(ws); W_SWEEP.wsch=ch end
   if ch then
     for i=1,#ch do
       local c=ch[i]
       local okn,nm=pcall(PF.name,c)
       if okn and nm==name then return c end
     end
   end
   return nil
end

local function w_push(list,inst,name,cid,fixed,extra,slow)
   local key=W_SWEEP.key(inst)
   local p=w_part(inst,key,extra)
   if not p then return end
   local x,y,z=w_pos(inst,p,fixed,key)
   if x then
     local pool=list.pool
     local e
     if pool and #pool>0 then e=pool[#pool]; pool[#pool]=nil end
     if not e then e={} end
     e.x=x; e.y=y; e.z=z; e.part=p
     e.live=not fixed; e.slow=slow or nil
     e.name=name; e.cid=cid; e.ph=#list%3
     e.dm=nil; e.dtxt=nil; e.dw=nil; e.dwsz=nil
     list[#list+1]=e
   end
end
