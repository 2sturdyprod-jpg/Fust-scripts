-- ---------- permissions ----------
local function getPerms()
    if S.perms == nil then
        local mod = Fust:FindFirstChild("AdminPerms")
        local ok, res
        if mod and mod:IsA("ModuleScript") then
            local m = require(mod)
            ok, res = pcall(m, mod)
        else
            ok, res = false, nil
        end
        if not ok or type(res) ~= "table" then res = false end
        S.perms = res
    end
    return S.perms or nil
end

    if not char then return end

    local hum = char:FindFirstChildOfClass("Humanoid")
    local isR15 = (hum and hum.RigType == Enum.HumanoidRigType.R15)
        or char:FindFirstChild("UpperTorso") ~= nil
    local bones = isR15 and R15_BONES or R6_BONES
    local camPos = camera.GetPosition()

    -- store projected endpoints; actual draw.Line calls happen in the render pass
    skel.points = {}
    for i, pair in ipairs(bones) do
        local p1 = char:FindFirstChild(pair[1])
        local p2 = char:FindFirstChild(pair[2])
        if p1 and p2 and utility.IsValid(p1) and utility.IsValid(p2) then
            local s1, on1 = draw.WorldToScreen(p1.Position.X, p1.Position.Y, p1.Position.Z)
            local s2, on2 = draw.WorldToScreen(p2.Position.X, p2.Position.Y, p2.Position.Z)
            if on1 and on2 then
                skel.points[i] = { s1, s2, color }
            end
        end
    end
end

-- ---------- body extents ----------
local function bodyExtents(root, char)
    local cf = root.CFrame
    local lo = Vector3.New(math.huge, math.huge, math.huge)
    local hi = -lo

    local function add(part)
        if part.Transparency < 1 then
            local localCf = cf:ToObjectSpace(part.CFrame)
            local half    = part.Size / 2
            local ex = math.abs(localCf.XVector.X)*half.X + math.abs(localCf.YVector.X)*half.Y + math.abs(localCf.ZVector.X)*half.Z
            local ey = math.abs(localCf.XVector.Y)*half.X + math.abs(localCf.YVector.Y)*half.Y + math.abs(localCf.ZVector.Y)*half.Z
            local ez = math.abs(localCf.XVector.Z)*half.X + math.abs(localCf.YVector.Z)*half.Y + math.abs(localCf.ZVector.Z)*half.Z
            local ext = Vector3.New(ex, ey, ez)
            lo = lo:Min(localCf.Position - ext)
            hi = hi:Max(localCf.Position + ext)
        end
    end

    for _, child in ipairs(char:GetChildren()) do
        if child:IsA("BasePart") then
            add(child)
        elseif child:IsA("Accessory") or child:IsA("Tool") then
            local handle = child:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then add(handle) end
        end
    end

    if lo.X > hi.X then
        return CFG.BODY_LO, CFG.BODY_HI
    end
    return lo:Max(Vector3.New(-8,-8,-8)), hi:Min(Vector3.New(8,8,8))
end

-- ---------- kind ----------
local function kind(uid, e)
    if e.staff then
        return COL.staff, (e.inv and "STAFF • INVIS" or "STAFF"), COL.staff
    elseif e.inv then
        return COL.muted, "INVIS", COL.hover
    else
        local specTarget = LocalPlayer:GetAttribute("FustSpecTarget")
        if type(specTarget) ~= "number" then specTarget = nil end
        if specTarget and uid == specTarget then
            return COL.amber,
                (e.watch and "SPECTATING • WATCH" or (e.flag and "SPECTATING • AC" or "SPECTATING")),
                COL.amber
        end

        local mateTeam
        local entry = S.entries[specTarget]
        if entry then mateTeam = entry.team end

        if mateTeam and e.team == mateTeam then
            return COL.mate,
                (e.watch and "TEAMMATE • WATCH" or (e.flag and "TEAMMATE • AC" or "TEAMMATE")),
                COL.mate
        elseif e.watch then
            return COL.watch, "WATCH", COL.watch
        elseif e.flag then
            return COL.bad, "AC", COL.badFill
        else
            return COL.text, nil, nil
        end
    end
end

-- ---------- speed text ----------
local function speedText(e, char, showSlow)
    local vel, seated
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hrp and hrp:IsA("BasePart") then vel = hrp.AssemblyLinearVelocity end
        seated = hum and hum.SeatPart ~= nil or false
    else
        seated = false
    end

    local speed
    if vel then
        speed = Vector3.New(vel.X, 0, vel.Z).Magnitude
    else
        speed = e.speed
    end
    if type(speed) ~= "number" or speed ~= speed then return "" end

    if (showSlow and (seated or speed < CFG.FAST_AMBER)) then return "" end

    local c
    if seated or speed < CFG.FAST_AMBER then c = COL.muted
    elseif speed < CFG.FAST_RED then c = COL.amber
    else c = COL.bad end

    return (" • <font color=\"#%s\">%d st/s%s</font>")
        :format(c:ToHex(), math.floor(speed + 0.5), seated and ", seated" or "")
end

-- ---------- fill mark ----------
local function fill(mark, rec, forceDetail, specUid, mateTeam, matePos)
    local detail = forceDetail
    if not detail then
        detail = rec.d > CFG.FAR + (mark.far and -CFG.FAR_HYST or CFG.FAR_HYST)
    end

    if mark.far ~= detail then
        mark.far = detail
        local fade = detail and CFG.FAR_FADE or 0
        mark.name.TextTransparency = fade
        mark.info.TextTransparency = fade
        mark.stroke.Transparency   = detail and 0.55 or 0.1
    end

    local kc, label, lc = kind(rec.uid, rec.e)
    mark.key = rec.key
    if mark.color ~= kc then
        mark.color = kc
        mark.name.TextColor3 = kc
        mark.stroke.Color = kc
        mark.pointer.BackgroundColor3 = kc
    end

    local ply = entity.GetPlayerByUserId(rec.uid)  -- helper below
    local displayName = ply and ply.DisplayName or ("User " .. rec.uid)

    if mark.name.Text ~= displayName then mark.name.Text = displayName end

    mark.tape.Visible = label ~= nil
    if label and lc then
        if mark.tape.Text ~= label then mark.tape.Text = label end
        mark.tape.BackgroundColor3 = lc
        mark.tape.TextColor3 = (lc == COL.hover) and COL.text or COL.slot
    end

    local hp = math.max(0, rec.e.h)
    local frac = (rec.e.m > 0) and math.clamp(hp / rec.e.m, 0, 1) or 0
    local hpColor
    if frac > 0.5 then hpColor = COL.good
    elseif frac > 0.25 then hpColor = COL.amber
    else hpColor = COL.bad end

    mark.fill.Size = UDim2.fromScale(1, frac)
    mark.fill.BackgroundColor3 = hpColor

    local dist = math.floor(rec.d + 0.5)
    local arrowText = ("%s • %d st"):format(displayName, dist)
    if mark.arrowLabel.Text ~= arrowText then mark.arrowLabel.Text = arrowText end

    if detail then
        local info = ("%d st"):format(dist)
        if mark.info.Text ~= info then mark.info.Text = info end
    else
        local near = forceDetail or rec.d <= CFG.DETAIL_NEAR
        local hex = hpColor:ToHex()
        local info = ("<font color=\"#%s\">%d HP</font> • %d st")
            :format(hex, math.floor(hp + 0.5), dist)
            .. speedText(rec.e, rec.char, not near)

        if forceDetail then
            if specUid and rec.uid ~= specUid and mateTeam ~= nil and rec.e.team == mateTeam then
                local mateHex = COL.mate:ToHex()
                local dFromMate = (rec.pos - matePos).Magnitude
                info = info .. (" • <font color=\"#%s\">%d st from them</font>")
                    :format(mateHex, math.floor(dFromMate + 0.5))
            end
            if rec.e.watch and rec.e.watchWhy then
                info = info .. (" • <font color=\"#%s\">%s</font>")
                    :format(COL.watch:ToHex(), rec.e.watchWhy)
            end
            if rec.e.watching then
                info = info .. (" • <font color=\"#%s\">watching %s</font>")
                    :format(COL.staff:ToHex(), rec.e.watching)
            end
            if rec.e.watchedBy and #rec.e.watchedBy > 0 then
                info = info .. (" • <font color=\"#%s\">watched by %s</font>")
                    :format(COL.staff:ToHex(), table.concat(rec.e.watchedBy, ", "))
            end
        end

        if mark.info.Text ~= info then mark.info.Text = info end
    end
end

-- ---------- highlights ----------
local function applyHighlights(records, specUid)
    local wanted = {}
    if LocalPlayer:GetAttribute("FustReplayOpen") ~= true then
        for _, r in ipairs(records) do
            if not r.char then continue end
            if r.uid == specUid then continue end
            if r.d <= CFG.HL_NEAR then continue end
            if not r.key and (r.uid ~= S.pick or r.d > CFG.HL_RANGE) then continue end

            local rank = ((r.key or r.uid == S.pick) and 0 or 2000000)
                + (r.onScreen and 0 or 1000000)
                + r.d
                - (S.hls[r.uid] and CFG.HL_KEEP_BONUS or 0)
            table.insert(wanted, { r = r, rank = rank })
        end
        table.sort(wanted, function(a, b) return a.rank < b.rank end)
    end

    local active = {}
    local folder = S.hlFolder
    if not folder or folder.Parent ~= workspace then
        folder = instance.New("Folder", workspace)
        folder.Name = "FustESPOutlines"
        S.hlFolder = folder
    end

    for i, w in ipairs(wanted) do
        if i > CFG.MAX_HL then break end
        local r = w.r
        local char = r.char
        local hl = S.hls[r.uid]
        if hl and hl.Parent ~= folder then
            instance.Destroy(hl)
            hl = nil
        end
        if not hl then
            hl = instance.New("Highlight", char)
            hl.Name = "Outline_" .. r.uid
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.FillTransparency = 0.85
            hl.OutlineTransparency = 0
            S.hls[r.uid] = hl
        end
        if hl.Adornee ~= char then hl.Adornee = char end

        local c = kind(r.uid, r.e)
        if hl.OutlineColor ~= c then
            hl.OutlineColor = c
            hl.FillColor = c
        end
        active[r.uid] = true
    end

    for uid, hl in pairs(S.hls) do
        if active[uid] then continue end
        if not hl then continue end
        instance.Destroy(hl)
        S.hls[uid] = nil
    end
end

-- ---------- pill ----------
local function setPill(waiting, count, specUid)
    local hud = PlayerGui:FindFirstChild("FustAdmin")
    if hud then hud = hud:FindFirstChild("Hud") end
    local pill = hud and hud:FindFirstChild("EspPill")
    if not (pill and pill:IsA("TextLabel")) then return end

    pill.Visible = true
    if waiting then
        pill.Text = "ESP, WAITING FOR THE SERVER"
        return
    end

    local text = ("ESP • %d PLAYERS • %s"):format(count, StaffKeys.label("esp"))
    local team = S.team
    if team and specUid and team.s == specUid then
        local here = 1
        local away = {}
        for _, m in ipairs(team.members) do
            if m.here then here = here + 1
            else table.insert(away, m.n) end
        end
        text = text .. (" • TEAM %d/%d HERE"):format(here, #team.members + 1)
        if #away > 0 then
            text = text .. " • AWAY: " .. string.upper(table.concat(away, ", "))
        end
    elseif specUid then
        text = text .. " • NO TEAM"
    end
    pill.Text = text
end

-- ---------- refresh ----------
local function refresh(now)
    local dt = now - S.lastPacket
    if dt > CFG.STALE_CLEAR and next(S.entries) ~= nil then
        clearAll()
    end

    if not ensureScreen() then return end

    -- seed missing players from entity cache
    for _, p in ipairs(entity.GetPlayers()) do
        if p.IsLocal then continue end
        if S.entries[p.UserId] then continue end
        local char = p.Character
        if not char then continue end
        local hum = p.Humanoid
        S.entries[p.UserId] = {
            pos   = p.Position or Vector3.New(0,0,0),
            vel   = p.Velocity or Vector3.New(0,0,0),
            at    = now,
            h     = p.Health,
            m     = p.MaxHealth,
            speed = (p.Velocity and (Vector3.New(p.Velocity.X, 0, p.Velocity.Z).Magnitude)) or 0,
            flag  = false, watch = false, inv = false, staff = false,
        }
    end

    local textUpdate = dt > (1 / CFG.TEXT_HZ)
    local camPos = camera.GetPosition()
    local sw, sh = draw.GetScreenSize()
    local center = Vector2.new(sw / 2, sh / 2)

    local specUid = LocalPlayer:GetAttribute("FustSpecTarget")
    if type(specUid) ~= "number" then specUid = nil end

    local mateEntry = specUid and S.entries[specUid] or nil
    local mateTeam  = mateEntry and mateEntry.team or nil

    local matePos
    if specUid then
        local mp = entity.GetPlayerByUserId(specUid)
        matePos = mp and mp.Position or nil
        if not matePos and S.entries[specUid] then
            matePos = S.entries[specUid].pos
        end
    end

    local pickPx = CFG.PICK_PX
    local records = {}
    local nearest, picked = nil, nil

    for uid, e in pairs(S.entries) do
        local p = entity.GetPlayerByUserId(uid)
        local char = p and p.Character or nil
        local root = char and char:FindFirstChild("HumanoidRootPart") or nil
        if not (root and utility.IsValid(root)) then
            root, char = nil, nil
        end

        local pos
        if root then
            pos = root.Position
        else
            pos = e.pos
            if pos then
                local age = now - (e.at or 0)
                if e.vel then
                    pos = pos + e.vel * math.clamp(age, 0, CFG.PREDICT)
                end
            end
        end
        if not pos then continue end

        local sx, sy, onScreen = draw.WorldToScreen(pos.X, pos.Y, pos.Z)
        local pxDist
        if onScreen then
            pxDist = (Vector2.new(sx, sy) - center).Magnitude
            if pxDist < pickPx then
                nearest, pickPx = uid, pxDist
            end
            if onScreen and uid == S.pick and pxDist < (CFG.PICK_PX * CFG.PICK_KEEP) then
                picked = uid
            end
        end

        local key = (uid == specUid or (mateTeam ~= nil and e.team == mateTeam))
            or e.flag or e.watch or e.inv or e.staff

        table.insert(records, {
            uid = uid, e = e, root = root, char = char,
            pos = pos, d = (pos - camPos).Magnitude,
            key = key, onScreen = onScreen,
        })
    end

    S.pick = picked or nearest

    for _, rec in ipairs(records) do
        local mark = S.marks[rec.uid] or makeMark(rec.uid)
        if not mark then continue end
        if rec.root and rec.char then
            local lo, hi = bodyExtents(rec.root, rec.char)
            mark.lo, mark.hi = lo, hi
        end
        fill(mark, rec, rec.key or rec.uid == S.pick, specUid, mateTeam, matePos)
    end

    -- drop stale marks
    for uid in pairs(S.marks) do
        if S.entries[uid] then continue end
        local m = S.marks[uid]
        if not m then continue end
        m.frame:Destroy()
        m.arrow:Destroy()
        S.marks[uid] = nil
        if S.skeletons[uid] then
            if S.skeletons[uid].folder then S.skeletons[uid].folder:Destroy() end
            S.skeletons[uid] = nil
        end
    end

    applyHighlights(records, specUid)
    setPill(not textUpdate, #records, specUid)
end

-- ---------- edge point ----------
local function edgePoint(cam, worldPos)
    local sw, sh = draw.GetScreenSize()
    local center = Vector2.new(sw / 2, sh / 2)
    local obj = cam.CFrame:PointToObjectSpace(worldPos)
    local dir = Vector2.new(obj.X, -obj.Y)
    if dir.Magnitude < 0.001 then dir = Vector2.New(0, 1) end
    local u = dir.Unit

    local xEdge = center.X - CFG.EDGE
    local yEdge = (u.Y < 0) and (center.Y - CFG.EDGE_TOP) or (center.Y - CFG.EDGE_BOTTOM)

    local tx, ty = math.huge, math.huge
    if math.abs(u.X) > 0.001 then tx = xEdge / math.abs(u.X) end
    if math.abs(u.Y) > 0.001 then ty = yEdge / math.abs(u.Y) end

    return center + u * math.min(tx, ty), u
end

-- ---------- per-frame render ----------
local function frame(dt)
    local now = utility.GetTime()

    if now - S.textAt >= 0.2 then
        S.textAt = now
        local ok, err = pcall(refresh, now)
        if not ok then warn("[Fust] FustAdminESP: refresh failed:", err) end
    end

    local cam = workspace.CurrentCamera
    if not cam then return end

    for uid, mark in pairs(S.marks) do
        local e = S.entries[uid]
        if not e then continue end

        local p = entity.GetPlayerByUserId(uid)
        local char = p and p.Character or nil
        local root = char and char:FindFirstChild("HumanoidRootPart") or nil
        if not (root and utility.IsValid(root)) then root = nil end

        local pos
        if root then
            pos = root.Position
        else
            pos = e.pos
            if pos then
                local age = now - (e.at or 0)
                if e.vel then
                    pos = pos + e.vel * math.clamp(age, 0, CFG.PREDICT)
                end
            end
        end
        if not pos then continue end

        -- snap correction
        if not root and mark.shown and (pos - mark.shown).Magnitude <= CFG.SNAP then
            local t = math.min(1, dt * (CFG.CORRECT / CFG.SNAP))
            pos = mark.shown:Lerp(pos, t)
        end
        mark.shown = pos

        local lo, hi = mark.lo, mark.hi
        local cf = root and root.CFrame or CFrame.new(pos)

        if not root then
            local maxAxis = math.max(-lo.X, hi.X, -lo.Z, hi.Z)
            lo = Vector3.New(-maxAxis, lo.Y, -maxAxis)
            hi = Vector3.New( maxAxis, hi.Y,  maxAxis)
        end

        local sw, sh = draw.GetScreenSize()
        local corners = 0
        local minX, minY = math.huge, math.huge
        local maxX, maxY = -math.huge, -math.huge

        for i = 0, 7 do
            local cx = (i % 2 == 0) and lo.X or hi.X
            local cy = (i % 4 < 2) and lo.Y or hi.Y
            local cz = (i < 4) and lo.Z or hi.Z
            local wx, wy, wz = cf:PointToWorldSpace(Vector3.New(cx, cy, cz)):Unpack()
            local sx, sy, ok = draw.WorldToScreen(wx, wy, wz)
            if not ok then continue end
            corners = corners + 1
            minX = math.min(minX, sx); maxX = math.max(maxX, sx)
            minY = math.min(minY, sy); maxY = math.max(maxY, sy)
        end

        if corners > 0
            and maxX >= 0 and minX <= sw
            and maxY >= 0 and minY <= sh then

            local x0 = math.max(minX, -sw)
            local x1 = math.min(maxX, sw + CFG.BOX_PAD)
            local y0 = math.max(minY, -sh)
            local y1 = math.min(maxY, sh + CFG.BOX_PAD)

            local bw = math.max((x1 - x0) + CFG.BOX_PAD, CFG.BOX_MIN)
            local bh = math.max((y1 - y0) + CFG.BOX_PAD, CFG.BOX_MIN)

            mark.frame.Position = UDim2.fromOffset((x0 + x1) / 2, y0 - CFG.BOX_PAD)
            mark.box.Size = UDim2.fromOffset(bw, bh)
            mark.frame.Visible = true
            mark.arrow.Visible = false

            if char then
                updateSkeleton(uid, char, mark.color)
            else
                hideSkeleton(uid)
            end
        else
            mark.frame.Visible = false
            hideSkeleton(uid)

            if mark.key then
                local ep, u = edgePoint(cam, pos)
                mark.arrow.Position = UDim2.fromOffset(ep.X, ep.Y)
                mark.pointer.Position = UDim2.fromOffset(u.X * 12, u.Y * 12)
                mark.arrowLabel.Position = UDim2.fromOffset(-u.X * 16, -u.Y * 16)
                mark.arrow.Visible = true
            else
                mark.arrow.Visible = false
            end
        end
    end

    drawSkeletons()
end

-- ---------- LOS lines module ----------
local LOS = {
    on = false, conn = nil, lines = {}, checkAt = 0, filterAt = -math.huge,
    params = nil, pitch = {}, lookHooked = false, SS = nil, folder = nil,

    shared = function()
        if LOS.SS == nil then
            local mod = Fust:FindFirstChild("SpecShared")
            local ok, res = pcall(function()
                if mod and mod:IsA("ModuleScript") then return require(mod) end
            end)
            if ok and type(res) == "table" then
                if type(res.losAt) ~= "function" then res = false end
                if type(res.K) ~= "table" or not res.K.LOS then res = false end
            else
                res = false
            end
            LOS.SS = res
        end
        return LOS.SS or nil
    end,

    hookLook = function()
        if LOS.lookHooked then return end
        LOS.lookHooked = true
        task.spawn(function()
            local ev = Remotes:WaitForChild("LookPitch", 30)
            local decMod = Fust:FindFirstChild("LookShared")
            if not (ev and ev:IsA("UnreliableRemoteEvent") and decMod and decMod:IsA("ModuleScript")) then return end
            local M = require(decMod)
            local ok, D = pcall(M, decMod)
            if not (ok and type(D) == "table") then return end
            if type(D.Decode) ~= "function" or type(D.ToRadians) ~= "function" then return end

            local conn = ev.OnClientEvent:Connect(function(data)
                if not LOS.on then return end
                local ok2, tbl = pcall(D.Decode, data)
                if not (ok2 and type(tbl) == "table") then return end
                local t = utility.GetTime()
                for k, v in pairs(tbl) do
                    if type(k) == "number" and type(v) == "number" then
                        LOS.pitch[k] = { r = D.ToRadians(v), at = t }
                    end
                end
            end)
            table.insert(S.conns, conn)
        end)
    end,

    specUid = function()
        local v = LocalPlayer:GetAttribute("FustSpecTarget")
        return type(v) == "number" and v or nil
    end,

    wanted = function()
        if LocalPlayer:GetAttribute("FustLosLines") == true
            and (S.on or LOS.specUid() ~= nil) then
            return StaffState.get(LocalPlayer, "StaffPvP") ~= true
        end
        return false
    end,

    aim = function(uid)
        local ss = LOS.shared()
        local e = S.entries[uid]
        local p = entity.GetPlayerByUserId(uid)
        local char = p and p.Character or nil
        local root = char and char:FindFirstChild("HumanoidRootPart") or nil
        if not (root and utility.IsValid(root)) then root, char = nil, nil end

        local pitchRec = LOS.pitch[uid]
        local pitch
        if pitchRec and utility.GetTime() - pitchRec.at < 2.5 then
            pitch = pitchRec.r
        elseif e and e.aim then
            pitch = math.asin(math.clamp(e.aim.Y, -1, 1))
        else
            pitch = 0
        end

        if root and char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health <= 0 then return nil, nil end
            local head = char:FindFirstChild("Head")
            local aimPt = (head and head:IsA("BasePart")) and head.Position
                or (root.Position + Vector3.New(0, 1.5, 0))
            local look = root.CFrame.LookVector
            return aimPt, ss.aimDir(look.X, look.Z, pitch)
        elseif e and e.pos and e.aim then
            return e.pos + Vector3.New(0, 1.5, 0), ss.aimDir(e.aim.X, e.aim.Z, pitch)
        end
        return nil, nil
    end,

    chest = function(uid)
        local e = S.entries[uid]
        if e and e.inv then return nil end
        local p = entity.GetPlayerByUserId(uid)
        local char = p and p.Character or nil
        local root = char and char:FindFirstChild("HumanoidRootPart") or nil
        if not (root and utility.IsValid(root)) then root, char = nil, nil end

        if root and char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health <= 0 then return nil end
            return root.Position + Vector3.New(0, 0.8, 0)
        elseif e and e.pos then
            return e.pos + Vector3.New(0, 0.8, 0)
        end
        return nil
    end,

    solid = function(origin, dir, maxDist)
        local params = LOS.params
        local travelled = 0
        for _ = 1, 4 do
            local remaining = maxDist - travelled
            if remaining <= 0.05 then return nil end
            local to = origin + dir * remaining
            local hit, pos, dist, inst, terrain = raycast.Cast(origin + dir * travelled, to)
            if not hit then return nil end
            local hitDist = travelled + dist
            local instOk = inst and inst:IsA("BasePart")
            if instOk then
                instOk = inst.Transparency >= 0.9 or not inst.CanCollide
            end
            if not instOk or inst == workspace.Terrain then return hitDist end
            travelled = hitDist + 0.05
        end
        return nil
    end,

    refreshFilter = function(now)
        if not LOS.params then
            LOS.params = RaycastParams.new()
            LOS.params.FilterType = Enum.RaycastFilterType.Exclude
            LOS.params.IgnoreWater = true
        end
        if now - LOS.filterAt >= 1 then
            LOS.filterAt = now
            local chars = {}
            for _, p in ipairs(entity.GetPlayers()) do
                if p.Character then table.insert(chars, p.Character) end
            end
            LOS.params.FilterDescendantsInstances = chars
        end
    end,

    pick = function(camPos, maxLines)
        local out = {}
        local spec = LOS.specUid()
        if spec and spec ~= LocalPlayer.UserId then table.insert(out, spec) end
        if S.on then
            local list = {}
            for uid, e in pairs(S.entries) do
                if uid == spec or e.inv or not e.pos then continue end
                table.insert(list, { uid = uid, d = (e.pos - camPos).Magnitude })
            end
            table.sort(list, function(a, b) return a.d < b.d end)
            for _, item in ipairs(list) do
                if #out >= maxLines + (spec and 1 or 0) then break end
                table.insert(out, item.uid)
            end
        end
        return out
    end,

    others = function(selfUid)
        local seen, out = {}, {}
        for uid in pairs(S.entries) do
            if uid == selfUid then continue end
            seen[uid] = true
            table.insert(out, uid)
        end
        for _, p in ipairs(entity.GetPlayers()) do
            if p.UserId == selfUid or seen[p.UserId] or not p.Character then continue end
            table.insert(out, p.UserId)
        end
        return out
    end,

    check = function(selfUid, origin, dir, len)
        local ss = LOS.shared()
        local solidDist = LOS.solid(origin, dir, len) or len
        local best, bestDist = nil, math.huge

        for _, uid in ipairs(LOS.others(selfUid)) do
            local chest = LOS.chest(uid)
            local d
            if chest then d = ss.losAt(origin, dir, chest, len) end
            if not d or d >= bestDist then continue end
            best, bestDist = chest, d
        end

        if not best then return solidDist, false end
        local delta = best - origin
        local mag = delta.Magnitude
        if mag < 1 then return solidDist, false end
        local block = LOS.solid(origin, delta.Unit, mag)
        local blocked = (block ~= nil) and (block < mag - 1.5) or false
        return solidDist, blocked
    end,

    line = function(uid)
        if not LOS.folder or LOS.folder.Parent ~= PlayerGui then
            for _, l in pairs(LOS.lines) do
                l.box:Destroy(); l.at:Destroy()
                LOS.lines[l.uid] = nil
            end
            local folder = instance.New("Folder", PlayerGui)
            folder.Name = "FustLosLines"
            LOS.folder = folder
        end
        if LOS.lines[uid] then return LOS.lines[uid] end

        local tpl = PlayerGui:FindFirstChild("FustAdminKit")
        if tpl then tpl = tpl:FindFirstChild("World") end
        if tpl then tpl = tpl:FindFirstChild("EspLos") end
        if not (tpl and tpl:IsA("BoxHandleAdornment")) then return nil end

        local box = tpl:Clone()
        box.Name = "Los_" .. uid
        local at = makeAnchor("FustLos_" .. uid)
        box.Adornee = at
        box.CFrame = CFrame.identity
        box.Parent = LOS.folder

        local rec = { uid = uid, box = box, at = at, len = 0, red = false }
        LOS.lines[uid] = rec
        return rec
    end,

    drop = function(uid)
        local l = LOS.lines[uid]
        if l then
            l.box:Destroy(); l.at:Destroy()
            LOS.lines[uid] = nil
        end
    end,

    step = function()
        local ss = LOS.shared()
        local cam = workspace.CurrentCamera
        if not (ss and cam) then return end

        local K = ss.K.LOS
        local now = utility.GetTime()
        local camPos = cam.CFrame.Position

        if now - LOS.checkAt >= 1 / K.HZ then
            LOS.checkAt = now
            LOS.refreshFilter(now)

            local active = {}
            for _, uid in ipairs(LOS.pick(camPos, K.MAX_LINES)) do
                local aimPt, aimDir = LOS.aim(uid)
                local line = (aimPt and aimDir) and LOS.line(uid) or nil
                if not (line and aimPt and aimDir) then continue end
                active[uid] = true
                local len, red = LOS.check(uid, aimPt, aimDir, K.LEN)
                line.len, line.red = len, red
            end
            for uid in pairs(LOS.lines) do
                if active[uid] then continue end
                LOS.drop(uid)
            end
        end

        for uid, l in pairs(LOS.lines) do
            local aimPt, aimDir = LOS.aim(uid)
            if aimPt and aimDir then
                local endPt = aimPt + aimDir * l.len
                local delta = endPt - aimPt
                local mag = delta.Magnitude
                if mag < 0.5 then
                    l.box.Visible = false
                else
                    local mid = aimPt + delta * 0.5
                    local dist = (mid - camPos).Magnitude
                    local thickness = math.clamp(dist * 0.0025, 0.05, l.red and 0.3 or 0.22)
                    l.box.Size = Vector3.New(thickness, thickness, mag)
                    l.at.CFrame = CFrame.lookAt(mid, endPt)
                    l.box.Color3 = l.red and COL.bad or COL.text
                    l.box.Transparency = l.red and 0.1 or 0.35
                    l.box.Visible = true
                end
            else
                l.box.Visible = false
            end
        end
    end,

    sync = function()
        local want = LOS.wanted()
        if want then want = LOS.shared() ~= nil end
        if want == LOS.on then return end
        LOS.on = want

        if want then
            LOS.hookLook()
            LOS.checkAt = 0
            LOS.conn = RunService.RenderStepped:Connect(function()
                local ok, err = pcall(LOS.step)
                if not ok then
                    warn("[Fust] FustAdminESP: LOS lines failed:", err)
                    LOS.on = false
                    if LOS.conn then LOS.conn:Disconnect(); LOS.conn = nil end
                end
            end)
        else
            if LOS.conn then LOS.conn:Disconnect(); LOS.conn = nil end
            for uid in pairs(LOS.lines) do LOS.drop(uid) end
            table.clear(LOS.pitch)
        end
    end,
}

-- ---------- toggle ----------
local function setOn(on)
    if on == S.on then return end
    S.on = on

    if on then
        S.lastPacket = utility.GetTime()
        S.textAt = 0
        if not S.bound then
            S.bound = true
            -- per-frame render via thread loop (Vector API has no BindToRenderStep)
            thread.Create(function()
                while S.on do
                    local dt = utility.GetDeltaTime()
                    local ok, err = pcall(frame, dt)
                    if not ok then
                        warn("[Fust] FustAdminESP: frame failed:", err)
                    end
                    sleep(0)
                end
            end)
        end
    else
        if S.bound then
            S.bound = false
        end
        clearAll()
        if S.screen then
            instance.Destroy(S.screen)
            S.screen = nil
        end
    end
end

local function toggleRequest()
    local newState = not S.on
    setOn(newState)
    local rf = Remotes:FindFirstChild("AdminAction")
    if rf and rf:IsA("RemoteFunction") then
        S.busy = true
        task.spawn(function()
            pcall(function()
                rf:InvokeServer("esp", { on = newState })
            end)
            S.busy = false
        end)
    end
end

-- ---------- packet handler ----------
local function onPacket(data)
    if not (S.on and type(data) == "table") then return end
    local list = data.list
    if type(list) ~= "table" then return end

    local now = utility.GetTime()
    S.lastPacket = now
    S.team = nil

    local team = data.team
    if type(team) == "table" then
        if type(team.s) == "number" and type(team.members) == "table" then
            local members = {}
            for _, m in ipairs(team.members) do
                if type(m) ~= "table" then continue end
                if type(m.u) ~= "number" then continue end
                table.insert(members, {
                    u = m.u,
                    n = tostring(m.n or m.u),
                    here = m.here == true,
                })
            end
            S.team = { s = team.s, members = members }
        end
    end

    local seen = {}
    for _, item in ipairs(list) do
        if type(item) ~= "table" then continue end
        if type(item.u) ~= "number" then continue end
        if typeof(item.p) ~= "Vector3" or item.u == LocalPlayer.UserId then continue end
        seen[item.u] = true

        local e = S.entries[item.u]
        if not e then
            e = { h = 0, m = 0, inv = false, flag = false, staff = false, watch = false }
            S.entries[item.u] = e
        end

        local vel
        if typeof(item.v) == "Vector3" and item.v.Magnitude < 2000 then vel = item.v end
        local speed = vel and Vector3.New(vel.X, 0, vel.Z).Magnitude or nil

        e.speed = speed
        e.vel   = vel
        e.at    = now
        e.pos   = item.p
        e.h     = type(item.h) == "number" and item.h or 0
        e.m     = type(item.m) == "number" and item.m or 0
        e.inv   = item.i == true
        e.flag  = item.f == true
        e.staff = item.s == true
        e.watch = item.wt == true

        if type(item.wy) == "string" then
            e.watchWhy = (string.sub(item.wy, 1, 60))
                :gsub("[<>&]", { ["<"] = "&lt;", [">"] = "&gt;", ["&"] = "&amp;" })
        else
            e.watchWhy = nil
        end

        local aim
        if typeof(item.a) == "Vector3" and item.a.Magnitude > 0.5 then
            aim = item.a.Unit
        end
        e.aim = aim
        e.team = type(item.t) == "string" and item.t or nil

        local watchedBy
        if type(item.w) == "table" then
            watchedBy = {}
            for _, n in ipairs(item.w) do
                if type(n) == "string" then table.insert(watchedBy, n) end
            end
            if #watchedBy <= 0 then watchedBy = nil end
        end
        e.watchedBy = watchedBy
        e.watching = type(item.sp) == "string" and item.sp or nil
    end

    for uid, e in pairs(S.entries) do
        if seen[uid] or now - (e.at or -math.huge) <= CFG.ENTRY_KEEP then continue end
        S.entries[uid] = nil
    end
end

        for _, c in ipairs(S.conns) do c:Disconnect() end
        table.clear(S.conns)
        if LOS.conn then LOS.conn:Disconnect(); LOS.conn = nil end
    end)
    table.insert(S.conns, conn)
end

-- remote event listener
task.spawn(function()
    local ev = Remotes:WaitForChild("AdminESP", 120)
    if ev and ev:IsA("RemoteEvent") then
        local conn = ev.OnClientEvent:Connect(onPacket)
        table.insert(S.conns, conn)
    end
end)
