-- ============================ HELPERS ============================
local BONE_NAMES = { "Head", "UpperTorso", "HumanoidRootPart" }

local function get_bone(char, idx)
    if not char then return nil end
    local name = BONE_NAMES[(idx or 0) + 1] or "Head"
    return char:FindFirstChild(name)
end

local function dist2(ax,ay,az, bx,by,bz)
    local dx,dy,dz = ax-bx, ay-by, az-bz
    return dx*dx+dy*dy+dz*dz
end

local function is_visible(target_pos)
    if not cfg.aim_vis_check then return true end
    local ok, vis = pcall(function()
        local cam = camera.GetPosition()
        return raycast.IsVisible(cam, target_pos)
    end)
    return ok and vis or true
end

-- ============================ SILENT AIM ============================
local silent_active = false
local function sync_silent_aim()
    local want = cfg.aim_enabled and cfg.aim_silent
    if want ~= silent_active then
        if want then
            pcall(function() raycast.EnableSilentHook() end)
        else
            pcall(function() raycast.DisableSilentHook() end)
        end
        silent_active = want
    end
end

-- ============================ AIMBOT CORE ============================
local aim_target = nil
local last_fire_time = 0
local rapid_next = 0

local function pick_aim_target()
    local cam = camera.GetPosition()
    local best_score = math.huge
    aim_target = nil
    local fov2 = cfg.aim_fov * cfg.aim_fov

    for _, p in ipairs(entity.GetPlayers()) do
        if not p.IsLocal and p.IsAlive then
            -- team check via cached Team property
            if cfg.aim_team_check then
                local lp = entity.GetLocalPlayer()
                if lp and p.HasTeam and lp.HasTeam and p.Team == lp.Team then
                    continue
                end
            end
            local bone = get_bone(p.Character, cfg.aim_bone)
            local pos = bone and bone.Position or p.Position
            if not pos then continue end

            local d2 = dist2(pos.X,pos.Y,pos.Z, cam.X,cam.Y,cam.Z)
            if d2 > cfg.player_maxdist^2 then continue end
            if not is_visible(pos) then continue end

            local sx, sy, on = utility.WorldToScreen(pos)
            if on then
                local cx, cy = input.GetScreenCenter()
                local sdx, sdy = sx - cx, sy - cy
                local sd2 = sdx*sdx + sdy*sdy
                if sd2 < fov2 then
                    local score = sd2 + d2 * 0.001
                    if score < best_score then
                        best_score = score
                        aim_target = { player = p, bone = bone, pos = pos }
                    end
                end
            end
        end
    end
end

local function apply_aimbot()
    if not cfg.aim_enabled then
        aim_target = nil
        return
    end
    pick_aim_target()
    if not aim_target then return end

    local target = aim_target.pos

    -- Lead prediction using velocity
    if cfg.aim_prediction and aim_target.player.Velocity then
        local v = aim_target.player.Velocity
        local cam = camera.GetPosition()
        local d = (target - cam).Magnitude
        local t = d / math.max(1, cfg.aim_speed)
        target = target + (v * t)
    end

    -- Smoothed camera look
    local smooth = math.max(1, cfg.aim_smooth)
    pcall(function() camera.LookAt(target, smooth) end)

    -- Silent aim redirect
    if silent_active and aim_target.bone then
        pcall(function() raycast.SetSilentTarget(aim_target.bone) end)
    end

    -- Triggerbot
    if cfg.gun_trigger then
        local now = utility.GetTime()
        if now - last_fire_time >= (cfg.gun_trigger_ms/1000) then
            local sx, sy, on = utility.WorldToScreen(target)
            if on then
                local cx, cy = input.GetScreenCenter()
                local sdx, sdy = sx - cx, sy - cy
                if (sdx*sdx + sdy*sdy) < (cfg.aim_fov^2) then
                    utility.MouseClick("left")
                    last_fire_time = now
                end
            end
        end
    end

    -- Rapid fire
    if cfg.gun_rapid then
        local now = utility.GetTime()
        if now >= rapid_next then
            utility.MouseClick("left")
            rapid_next = now + (cfg.gun_rapid_ms/1000)
        end
    end
end

-- ============================ ESP RENDERING ============================
local function draw_player_esp()
    if not cfg.player_esp then return end
    local maxd2 = cfg.player_maxdist^2

    for _, p in ipairs(entity.GetPlayers()) do
        if not p.IsLocal and p.IsAlive then
            local d = p:DistanceTo()
            if d > cfg.player_maxdist then continue end

            local b = p:GetBounds()
            if not b.valid then
                if cfg.offscreen_ply then
                    local sw, sh = utility.GetScreenSize()
                    local sx, sy, _ = p:GetBoneScreen("Head")
                    local ang = math.atan2(sy - sh/2, sx - sw/2)
                    local ex = sw/2 + math.cos(ang) * (sw/2 - 20)
                    local ey = sh/2 + math.sin(ang) * (sh/2 - 20)
                    draw.Text(ex, ey, "< " .. p.Name, {1,0.4,0.4,1}, cfg.player_font)
                end
                continue
            end

            local col = menu.GetColor("player_box")

            if cfg.player_box or cfg.player_chams then
                if cfg.player_chams then
                    draw.RectFilled(b.x, b.y, b.w, b.h, col)
                end
                draw.Rect(b.x, b.y, b.w, b.h, col)
            end

            if cfg.player_health then
                local hp = p.Health
                local mhp = p.MaxHealth or 100
                local frac = math.min(1, math.max(0, hp/mhp))
                local bw = math.max(30, b.w)
                draw.RectFilled(b.x, b.y - 8, bw, 4, {0.1,0.1,0.1,0.8})
                draw.RectFilled(b.x, b.y - 8, bw*frac, 4, {frac>0.5 and 0.2 or 1, frac>0.5 and 1 or 0.3, 0.2, 1})
                draw.Text(b.x + bw/2, b.y - 20, tostring(math.floor(hp)), {1,1,1,1}, 10)
            end

            if cfg.player_name then
                draw.Text(b.x + b.w/2, b.y + b.h + 4, p.Name, {1,1,1,1}, cfg.player_font)
            end
            if cfg.player_dist then
                draw.Text(b.x + b.w/2, b.y + b.h + 18, string.format("%.0fm", d), {0.8,0.8,0.8,1}, 10)
            end
            if cfg.player_tracer then
                local tc = menu.GetColor("player_tracer")
                local cx, cy = input.GetScreenCenter()
                draw.Line(cx, cy, b.x + b.w/2, b.y + b.h/2, tc)
            end
        end
    end
end

local function draw_world_esp(kind, maxdist, font)
    local ws = game.Workspace
    if not ws then return end
    local cam = camera.GetPosition()
    local maxd2 = maxdist^2

    for _, obj in ipairs(ws:GetChildren()) do
        local n = (obj.Name or ""):lower()
        local match = false
        if kind == "loot" then
            match = (n:find("barrel") and cfg.loot_barrels) or
                    (n:find("crate")  and (cfg.loot_crates or cfg.loot_military or cfg.loot_elite)) or
                    (n:find("largebox") and cfg.loot_largebox) or
                    (n:find("smallbox") and cfg.loot_smallbox) or
                    (n:find("toolcup") and cfg.loot_tc) or
                    (n:find("locker") and cfg.loot_locker) or
                    (n:find("furnace") and cfg.loot_furnace) or
                    (n:find("workbench") and cfg.loot_workbench) or
                    (n:find("shelf") and cfg.loot_shelves) or
                    (n:find("sleeping") and cfg.loot_sleepbag)
        elseif kind == "bodybag" then
            match = n:find("bodybag") or n:find("corpse")
        elseif kind == "dropped" then
            match = n:find("dropped") or n:find("pickup")
        elseif kind == "node" then
            match = (n:find("stone") and cfg.node_stone) or
                    (n:find("sulfur") and cfg.node_sulfur) or
                    ((n:find("iron") or n:find("metal")) and cfg.node_iron)
        elseif kind == "pickup" then
            match = (n:find("ore") and cfg.pickup_ores) or
                    ((n:find("plant") or n:find("hemp") or n:find("corn") or n:find("pumpkin")) and cfg.pickup_plants) or
                    ((n:find("wood") or n:find("log")) and cfg.pickup_wood)
        end
        if not match then continue end

        local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart")
        if not part or not part.Position then continue end
        local pos = part.Position
        local d2 = dist2(pos.X,pos.Y,pos.Z, cam.X,cam.Y,cam.Z)
        if d2 > maxd2 then continue end

        local sx, sy, on = utility.WorldToScreen(pos)
        if not on then continue end

        local col = {1,1,1,1}
        if kind == "loot" then col = {0.95,0.7,0.25,1} end
        if kind == "bodybag" then col = {0.9,0.25,0.35,1} end
        if kind == "dropped" then col = {1,0.4,0.9,1} end
        if kind == "node" then col = {0.85,0.85,0.9,1} end
        if kind == "pickup" then col = {1,0.8,0.35,1} end

        draw.Text(sx, sy, obj.Name, col, font)
        if (kind=="loot" and cfg.loot_dist) or (kind=="bodybag" and cfg.bodybag_dist) or (kind=="dropped" and cfg.dropped_dist) or (kind=="node" and cfg.nodes_dist) or (kind=="pickup" and cfg.pickups_dist) then
            draw.Text(sx, sy+12, string.format("%.0fm", math.sqrt(d2)), {0.8,0.8,0.8,1}, 10)
        end
        if (kind=="loot" and cfg.loot_tracer) or (kind=="bodybag" and cfg.bodybag_tracer) or (kind=="dropped" and cfg.dropped_tracer) or (kind=="node" and cfg.nodes_tracer) then
            local cx, cy = input.GetScreenCenter()
            draw.Line(cx, cy, sx, sy, {1,1,1,0.35})
        end
    end
end

-- ============================ ENVIRONMENT ============================
local function apply_environment()
    if cfg.no_fog then
        lighting.SetFogStart(100000)
        lighting.SetFogEnd(100000)
    end
    if cfg.no_grass then
        terrain.SetGrassLength(0)
    end
    if cfg.bright_light then
        lighting.SetBrightness(2)
        lighting.SetAmbient(1)
        lighting.SetOutdoorAmbient(1)
    end
    if cfg.fov_override > 0 then
        camera.SetFov(cfg.fov_override)
    elseif cfg.zoom_enabled then
        camera.SetFov(cfg.zoom_fov)
    end
end

    end
