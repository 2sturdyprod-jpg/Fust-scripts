local function clamp(v,lo,hi) return v<lo and lo or (v>hi and hi or v) end
local function mix(c1,c2,t) return {c1[1]+(c2[1]-c1[1])*t,c1[2]+(c2[2]-c1[2])*t,c1[3]+(c2[3]-c1[3])*t,(c1[4] or 1)+((c2[4] or 1)-(c1[4] or 1))*t} end
local function ease_out(t) t=clamp(t,0,1); local f=1-t; return 1-f*f*f end
local function hsv(h,s,v) h=(h%1)*6; local i=math.floor(h); local f=h-i; local p=v*(1-s); local q=v*(1-s*f); local u=v*(1-s*(1-f))
if i==0 then return{v,u,p,1} elseif i==1 then return{q,v,p,1} elseif i==2 then return{p,v,u,1}
elseif i==3 then return{p,q,v,1} elseif i==4 then return{u,p,v,1} else return{v,p,q,1} end end

local _dt=0.016
local M_PER_STUD=0.28
local anim={}
local FRAME=0
local function approach(k,target,spd) local v=anim[k]; if v==nil then v=target end
v=v+(target-v)*clamp(_dt*(spd or 12),0,1); anim[k]=v; return v end

local SCR={w=1920,h=1080,f=-1}
local function scr()
    if SCR.f~=FRAME then
        local w,h=utility.GetScreenSize()
        SCR.w,SCR.h,SCR.f=w or 1920,h or 1080,FRAME
    end
    return SCR.w,SCR.h
end

local function mouse_pos() local x,y=utility.GetMousePos(); return x or 0, y or 0 end

local NOW={v=0,f=-1}
local function now()
    if NOW.f~=FRAME then
        NOW.v=utility.GetTime and utility.GetTime() or os.clock()
        NOW.f=FRAME
    end
    return NOW.v
end

local function ekey(vk)
    local ok,r=pcall(input.IsKeyDown,vk)
    if not ok or r==nil or r==false or r==0 then return false end
    return true
end

CFG.save=function(slot,quiet)
slot=tonumber(slot) or tonumber(cfg.cfg_slot) or 1
local data=nil
local hs=CFG.http()
if hs then
local t={v={},b={},c={}}
for k,v in pairs(cfg) do if k~="cfg_slot" and k:sub(1,7)~="gun_sgc" then t.v[k]=v end end
for k,v in pairs(bind) do t.b[k]=v end
for k,cv in pairs(col) do t.c[k]={cv[1] or 0,cv[2] or 0,cv[3] or 0,cv[4] or 1} end
local ok,s=pcall(function() return hs:JSONEncode(t) end)
if ok and type(s)=="string" and #s>0 then data=s end
end
if not data then
local o={"#bustui-config-v1"}
for k,v in pairs(cfg) do
if k~="cfg_slot" and k:sub(1,7)~="gun_sgc" then o[#o+1]="v "..k.."="..(type(v)=="boolean" and (v and "1" or "0") or tostring(v)) end
end
for k,v in pairs(bind) do o[#o+1]="b "..k.."="..tostring(v) end
for k,cv in pairs(col) do o[#o+1]="c "..k.."="..(cv[1] or 0)..","..(cv[2] or 0)..","..(cv[3] or 0)..","..(cv[4] or 1) end
data=table.concat(o,"\n")
end
local g=CFG.env()
if g then g.BUSTUI_CFGS[slot]=data; g.BUSTUI_SLOT=slot end
local disk=false
local ok,f=pcall(function() return io.open("bustui_config_"..slot..".cfg","w") end)
if ok and f then
local wok=pcall(function() f:write(data) end)
pcall(function() f:close() end)
disk=wok
end
cfg_dirty=false
if not quiet then
notify.Success("Config Saved","Slot "..slot,2)
end
end

CFG.load=function(slot,quiet)
slot=tonumber(slot) or 1
local g=CFG.env()
local data=g and g.BUSTUI_CFGS[slot]
if not data then
local ok,f=pcall(function() return io.open("bustui_config_"..slot..".cfg","r") end)
if ok and f then
local content=f:read("*a")
pcall(function() f:close() end)
data=content
end
end
if not data then return false end

for line in data:gmatch("[^\r\n]+") do
if line:sub(1,2)=="v " then
local k,v=line:match("^v ([^=]+)=(.+)$")
if k and v then
if v=="1" then cfg[k]=true
elseif v=="0" then cfg[k]=false
else cfg[k]=tonumber(v) or v end
end
elseif line:sub(1,2)=="b " then
local k,v=line:match("^b ([^=]+)=(.+)$")
if k and v then bind[k]=tonumber(v) end
elseif line:sub(1,2)=="c " then
local k,v=line:match("^c ([^=]+)=(.+)$")
if k and v then
local r,g2,b,a=v:match("([%d%.]+),([%d%.]+),([%d%.]+),([%d%.]+)")
if r then col[k]={tonumber(r),tonumber(g2),tonumber(b),tonumber(a)} end
end
end
end

-- Sync all values into the native menu
for k,v in pairs(cfg) do
pcall(function() menu.Set(k,v) end)
end
for k,v in pairs(bind) do
pcall(function() menu.SetKey(k,v) end)
end
for k,cv in pairs(col) do
pcall(function() menu.SetColor(k,cv) end)
end

refresh_all()

if not quiet then
notify.Success("Config Loaded","Slot "..slot,2)
end
return true
end

local function build_group(tab_name, group)
for _,it in ipairs(group.items) do
local id=it.id
local label=it.label
local parent=infer_parent(it)
local opts={}
if parent then opts.parent=parent end

if it.t=="toggle" then
menu.AddCheckbox(tab_name,group.name,id,label,cfg[id] or false,opts)
menu.SetCallback(id,function(v)
cfg[id]=v
cfg_dirty=true
refresh_all()
end)

elseif it.t=="slider" then
local def=cfg[id] or it.default or it.min
if math.floor(it.min)==it.min and math.floor(it.max)==it.max then
menu.AddSliderInt(tab_name,group.name,id,label,it.min,it.max,def,"%d")
else
menu.AddSliderFloat(tab_name,group.name,id,label,it.min,it.max,def,"%.2f")
end
menu.SetCallback(id,function(v)
cfg[id]=v
cfg_dirty=true
end)

elseif it.t=="combo" then
local idx=(cfg[id] or it.default or 1)-1
menu.AddCombo(tab_name,group.name,id,label,it.items,idx,opts)
menu.SetCallback(id,function(new_idx)
cfg[id]=new_idx+1
cfg_dirty=true
if id=="cfg_slot" then
if CFG.load(cfg[id],true) then
notify.Success("Slot "..cfg[id].." loaded","",2)
else
notify.Warning("Slot "..cfg[id].." is empty","settings are unchanged",2)
end
end
refresh_all()
end)

elseif it.t=="bind" then
menu.AddHotkey(tab_name,group.name,id,label,bind[id] or 0,opts)

elseif it.t=="color" then
local cc=col[id] or {1,1,1,1}
menu.AddColorpicker(tab_name,group.name,id,label,cc,opts)

elseif it.t=="button" then
menu.AddButton(tab_name,group.name,id,label,function()
if it._cb then it._cb() end
if id=="cfg_reset" then CFG.reset() end
if id=="unload" then
thread.StopAll()
OnFrame=nil
notify.Success("BustUI unloaded","",2)
end
end)

elseif it.t=="note" then
menu.AddLabel(tab_name,group.name,label)
end

ALL_ITEMS[#ALL_ITEMS+1]=it
end
end

local function esp_mate(p,lp)
if not p then return false end
local id=p.UserId
if not id then return false end
if lp and lp.UserId==id then return true end
if TEAM.roster[id] then return true end
if FRAME>=TEAM.next then
TEAM.next=FRAME+60
local ids=TEAM.ids
for k in pairs(ids) do ids[k]=nil end
local tf
pcall(function()
local ws
if game then
ws=game.Workspace or game.workspace
if not ws and game.GetService then ws=game:GetService("Workspace") end
end
if not ws then ws=workspace end
if not ws then return end
local cw=TEAM.child(ws,"CurrentCamera")
if not cw then cw=TEAM.child(ws,"Camera") end
tf=TEAM.child(cw,"FustTeamTags")
if not tf then tf=TEAM.child(ws,"FustTeamTags") end
if tf then
local list
local fn=tf.GetChildren
if type(fn)=="function" then
local ok,v=pcall(fn,tf)
if ok and type(v)=="table" then list=v end
end
if list then
for _,ch in ipairs(list) do
local nm=ch and ch.Name
if nm and nm:sub(1,3)=="Tag" then
local uid=tonumber(nm:sub(4))
if uid then ids[uid]=true end
end
end
else
if TEAM.child(tf,"Tag"..id) then ids[id]=true end
end
end
end)
TEAM.tf=tf
end
if TEAM.ids[id] then return true end
if TEAM.tf then return false end
if not (lp and lp.HasTeam and p.HasTeam) then return false end
return p.Team==lp.Team
end
